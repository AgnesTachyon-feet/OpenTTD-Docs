37NetworkRecvStatus NetworkAdminSocketHandler::HandlePacket(Packet &p)
38{
39 PacketAdminType type = (PacketAdminType)p.Recv_uint8();
40
41 if (this->HasClientQuit()) {
42 Debug(net, 0, "[tcp/admin] Received invalid packet from '{}' ({})", this->admin_name, this->admin_version);
43 this->CloseConnection();
44 return NETWORK_RECV_STATUS_MALFORMED_PACKET;
45 }
46
47 switch (type) {
48 case ADMIN_PACKET_ADMIN_JOIN: return this->Receive_ADMIN_JOIN(p);
49 case ADMIN_PACKET_ADMIN_QUIT: return this->Receive_ADMIN_QUIT(p);
50 case ADMIN_PACKET_ADMIN_UPDATE_FREQUENCY: return this->Receive_ADMIN_UPDATE_FREQUENCY(p);
51 case ADMIN_PACKET_ADMIN_POLL: return this->Receive_ADMIN_POLL(p);
52 case ADMIN_PACKET_ADMIN_CHAT: return this->Receive_ADMIN_CHAT(p);
53 case ADMIN_PACKET_ADMIN_EXTERNAL_CHAT: return this->Receive_ADMIN_EXTERNAL_CHAT(p);
54 case ADMIN_PACKET_ADMIN_RCON: return this->Receive_ADMIN_RCON(p);
55 case ADMIN_PACKET_ADMIN_GAMESCRIPT: return this->Receive_ADMIN_GAMESCRIPT(p);
56 case ADMIN_PACKET_ADMIN_PING: return this->Receive_ADMIN_PING(p);
57 case ADMIN_PACKET_ADMIN_JOIN_SECURE: return this->Receive_ADMIN_JOIN_SECURE(p);
58 case ADMIN_PACKET_ADMIN_AUTH_RESPONSE: return this->Receive_ADMIN_AUTH_RESPONSE(p);
59
60 case ADMIN_PACKET_SERVER_FULL: return this->Receive_SERVER_FULL(p);
61 case ADMIN_PACKET_SERVER_BANNED: return this->Receive_SERVER_BANNED(p);
62 case ADMIN_PACKET_SERVER_ERROR: return this->Receive_SERVER_ERROR(p);
63 case ADMIN_PACKET_SERVER_PROTOCOL: return this->Receive_SERVER_PROTOCOL(p);
64 case ADMIN_PACKET_SERVER_WELCOME: return this->Receive_SERVER_WELCOME(p);
65 case ADMIN_PACKET_SERVER_NEWGAME: return this->Receive_SERVER_NEWGAME(p);
66 case ADMIN_PACKET_SERVER_SHUTDOWN: return this->Receive_SERVER_SHUTDOWN(p);
67
68 case ADMIN_PACKET_SERVER_DATE: return this->Receive_SERVER_DATE(p);
69 case ADMIN_PACKET_SERVER_CLIENT_JOIN: return this->Receive_SERVER_CLIENT_JOIN(p);
70 case ADMIN_PACKET_SERVER_CLIENT_INFO: return this->Receive_SERVER_CLIENT_INFO(p);
71 case ADMIN_PACKET_SERVER_CLIENT_UPDATE: return this->Receive_SERVER_CLIENT_UPDATE(p);
72 case ADMIN_PACKET_SERVER_CLIENT_QUIT: return this->Receive_SERVER_CLIENT_QUIT(p);
73 case ADMIN_PACKET_SERVER_CLIENT_ERROR: return this->Receive_SERVER_CLIENT_ERROR(p);
74 case ADMIN_PACKET_SERVER_COMPANY_NEW: return this->Receive_SERVER_COMPANY_NEW(p);
75 case ADMIN_PACKET_SERVER_COMPANY_INFO: return this->Receive_SERVER_COMPANY_INFO(p);
76 case ADMIN_PACKET_SERVER_COMPANY_UPDATE: return this->Receive_SERVER_COMPANY_UPDATE(p);
77 case ADMIN_PACKET_SERVER_COMPANY_REMOVE: return this->Receive_SERVER_COMPANY_REMOVE(p);
78 case ADMIN_PACKET_SERVER_COMPANY_ECONOMY: return this->Receive_SERVER_COMPANY_ECONOMY(p);
79 case ADMIN_PACKET_SERVER_COMPANY_STATS: return this->Receive_SERVER_COMPANY_STATS(p);
80 case ADMIN_PACKET_SERVER_CHAT: return this->Receive_SERVER_CHAT(p);
81 case ADMIN_PACKET_SERVER_RCON: return this->Receive_SERVER_RCON(p);
82 case ADMIN_PACKET_SERVER_CONSOLE: return this->Receive_SERVER_CONSOLE(p);
83 case ADMIN_PACKET_SERVER_CMD_NAMES: return this->Receive_SERVER_CMD_NAMES(p);
84 case ADMIN_PACKET_SERVER_CMD_LOGGING: return this->Receive_SERVER_CMD_LOGGING(p);
85 case ADMIN_PACKET_SERVER_RCON_END: return this->Receive_SERVER_RCON_END(p);
86 case ADMIN_PACKET_SERVER_PONG: return this->Receive_SERVER_PONG(p);
87 case ADMIN_PACKET_SERVER_AUTH_REQUEST: return this->Receive_SERVER_AUTH_REQUEST(p);
88 case ADMIN_PACKET_SERVER_ENABLE_ENCRYPTION: return this->Receive_SERVER_ENABLE_ENCRYPTION(p);
89
90 default:
91 Debug(net, 0, "[tcp/admin] Received invalid packet type {} from '{}' ({})", type, this->admin_name, this->admin_version);
92 this->CloseConnection();
93 return NETWORK_RECV_STATUS_MALFORMED_PACKET;
94 }
95}

104NetworkRecvStatus NetworkAdminSocketHandler::ReceivePackets()
105{
106 std::unique_ptr<Packet> p;
107 while ((p = this->ReceivePacket()) != nullptr) {
108 NetworkRecvStatus res = this->HandlePacket(*p);
109 if (res != NETWORK_RECV_STATUS_OKAY) return res;
110 }
111
112 return NETWORK_RECV_STATUS_OKAY;
113}
