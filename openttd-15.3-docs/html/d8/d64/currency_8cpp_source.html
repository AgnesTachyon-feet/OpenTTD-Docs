128uint64_t GetMaskOfAllowedCurrencies()
129{
130 uint64_t mask = 0LL;
131 uint i;
132
133 for (i = 0; i < CURRENCY_END; i++) {
134 TimerGameCalendar::Year to_euro = _currency_specs[i].to_euro;
135
136 if (to_euro != CF_NOEURO && to_euro != CF_ISEURO && TimerGameCalendar::year >= to_euro) continue;
137 if (to_euro == CF_ISEURO && TimerGameCalendar::year < 2000) continue;
138 SetBit(mask, i);
139 }
140 SetBit(mask, CURRENCY_CUSTOM); // always allow custom currency
141 return mask;
142}

163void ResetCurrencies(bool preserve_custom)
164{
165 for (uint i = 0; i < CURRENCY_END; i++) {
166 if (preserve_custom && i == CURRENCY_CUSTOM) continue;
