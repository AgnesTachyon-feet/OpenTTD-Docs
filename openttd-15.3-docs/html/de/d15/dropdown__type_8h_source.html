41 virtual void Draw(const Rect &full, const Rect &, bool, int, Colours bg_colour) const
42 {

46 TextColour GetColour(bool sel) const
47 {
48 if (this->shaded) return (sel ? TC_SILVER : TC_GREY) | TC_NO_SHADE;
