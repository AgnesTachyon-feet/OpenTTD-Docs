19void Blitter_8bppSimple::Draw(Blitter::BlitterParams *bp, BlitterMode mode, ZoomLevel zoom)
20{
21 const uint8_t *src, *src_line;
22 uint8_t *dst, *dst_line;
23
24 /* Find where to start reading in the source sprite */
25 src_line = (const uint8_t *)bp->sprite + (bp->skip_top * bp->sprite_width + bp->skip_left) * ScaleByZoom(1, zoom);
26 dst_line = (uint8_t *)bp->dst + bp->top * bp->pitch + bp->left;
27
28 for (int y = 0; y < bp->height; y++) {
29 dst = dst_line;
30 dst_line += bp->pitch;
31
32 src = src_line;
33 src_line += bp->sprite_width * ScaleByZoom(1, zoom);
34
35 for (int x = 0; x < bp->width; x++) {
36 uint colour = 0;
37
38 switch (mode) {
39 case BlitterMode::ColourRemap:
40 case BlitterMode::CrashRemap:
41 colour = bp->remap[*src];
42 break;
43
44 case BlitterMode::Transparent:
45 case BlitterMode::TransparentRemap:
46 if (*src != 0) colour = bp->remap[*dst];
47 break;
48
49 case BlitterMode::BlackRemap:
50 if (*src != 0) *dst = 0;
51 break;
52
53 default:
54 colour = *src;
55 break;
56 }
57 if (colour != 0) *dst = colour;
58 dst++;
59 src += ScaleByZoom(1, zoom);
60 }
61 }
62}

64Sprite *Blitter_8bppSimple::Encode(SpriteType, const SpriteLoader::SpriteCollection &sprite, SpriteAllocator &allocator)
65{
66 const auto &root_sprite = sprite.Root();
67 Sprite *dest_sprite;
68 dest_sprite = allocator.Allocate<Sprite>(sizeof(*dest_sprite) + static_cast<size_t>(root_sprite.height) * static_cast<size_t>(root_sprite.width));
69
70 dest_sprite->height = root_sprite.height;
71 dest_sprite->width = root_sprite.width;
72 dest_sprite->x_offs = root_sprite.x_offs;
73 dest_sprite->y_offs = root_sprite.y_offs;
74
75 /* Copy over only the 'remap' channel, as that is what we care about in 8bpp */
76 uint8_t *dst = reinterpret_cast<uint8_t *>(dest_sprite->data);
77 for (int i = 0; i < root_sprite.height * root_sprite.width; i++) {
78 dst[i] = root_sprite.data[i].m;
79 }
80
81 return dest_sprite;
82}
