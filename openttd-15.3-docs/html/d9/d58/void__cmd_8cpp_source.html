22static void DrawTile_Void(TileInfo *ti)
23{
24 /* If freeform edges are off, draw infinite water off the edges of the map. */
25 if (!_settings_game.construction.freeform_edges) {
26 DrawGroundSprite(SPR_FLAT_WATER_TILE + SlopeToSpriteOffset(ti->tileh), PAL_NONE);
27 } else {
28 DrawGroundSprite(SPR_FLAT_BARE_LAND + SlopeToSpriteOffset(ti->tileh), PALETTE_ALL_BLACK);
29 }
30}

33static int GetSlopePixelZ_Void(TileIndex, uint x, uint y, bool)
34{
35 /* This function may be called on tiles outside the map, don't assume
36 * that 'tile' is a valid tile index. See GetSlopePixelZOutsideMap. */
37 auto [tileh, z] = GetTilePixelSlopeOutsideMap(x >> 4, y >> 4);
38
39 return z + GetPartialPixelZ(x & 0xF, y & 0xF, tileh);
40}
