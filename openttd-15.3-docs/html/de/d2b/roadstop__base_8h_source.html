159 inline uint AllocateBay()
160 {
161 assert(this->HasFreeBay());
162
163 /* Find the first free bay. */
164 uint bay_nr = 0;
165 while (!this->IsFreeBay(bay_nr)) ++bay_nr;
166
167 this->AllocateDriveThroughBay(bay_nr);
168 return bay_nr;
169 }
