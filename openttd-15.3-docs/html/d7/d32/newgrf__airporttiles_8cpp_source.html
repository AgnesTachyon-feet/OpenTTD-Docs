69void AirportTileOverrideManager::SetEntitySpec(AirportTileSpec &&airpts)
70{
71 StationGfx airpt_id = this->AddEntityID(airpts.grf_prop.local_id, airpts.grf_prop.grfid, airpts.grf_prop.subst_id);
72
73 if (airpt_id == this->invalid_id) {
74 GrfMsg(1, "AirportTile.SetEntitySpec: Too many airport tiles allocated. Ignoring.");
75 return;
76 }
77
78 AirportTileSpec::tiles[airpt_id] = std::move(airpts);
79
80 /* Now add the overrides. */
81 for (int i = 0; i < this->max_offset; i++) {
82 AirportTileSpec *overridden_airpts = &AirportTileSpec::tiles[i];
83
84 if (this->entity_overrides[i] != AirportTileSpec::tiles[airpt_id].grf_prop.local_id || this->grfid_overrides[i] != AirportTileSpec::tiles[airpt_id].grf_prop.grfid) continue;
85
86 overridden_airpts->grf_prop.override_id = airpt_id;
87 overridden_airpts->enabled = false;
88 this->entity_overrides[i] = this->invalid_id;
89 this->grfid_overrides[i] = 0;
90 }
91}

112static uint32_t GetNearbyAirportTileInformation(uint8_t parameter, TileIndex tile, StationID index, bool grf_version8)
113{
114 if (parameter != 0) tile = GetNearbyTile(parameter, tile); // only perform if it is required
115 bool is_same_airport = (IsTileType(tile, MP_STATION) && IsAirport(tile) && GetStationIndex(tile) == index);

117 return GetNearbyTileInformation(tile, grf_version8) | (is_same_airport ? 1 : 0) << 8;

129static uint32_t GetAirportTileIDAtOffset(TileIndex tile, const Station *st, uint32_t cur_grfid)
130{
131 if (!st->TileBelongsToAirport(tile)) {
132 return 0xFFFF;
133 }
134
135 StationGfx gfx = GetAirportGfx(tile);
136 const AirportTileSpec *ats = AirportTileSpec::Get(gfx);
137
138 if (gfx < NEW_AIRPORTTILE_OFFSET) { // Does it belongs to an old type?
139 /* It is an old tile. We have to see if it's been overridden */
140 if (ats->grf_prop.override_id == INVALID_AIRPORTTILE) { // has it been overridden?
141 return 0xFF << 8 | gfx; // no. Tag FF + the gfx id of that tile
142 }
143 /* Overridden */
144 const AirportTileSpec *tile_ovr = AirportTileSpec::Get(ats->grf_prop.override_id);
145
146 if (tile_ovr->grf_prop.grfid == cur_grfid) {
147 return tile_ovr->grf_prop.local_id; // same grf file
148 } else {
149 return 0xFFFE; // not the same grf file
150 }
151 }
152 /* Not an 'old type' tile */
153 if (ats->grf_prop.HasSpriteGroups()) {
154 if (ats->grf_prop.grfid == cur_grfid) { // same airport, same grf ?
155 return ats->grf_prop.local_id;
156 } else {
157 return 0xFFFE; // Defined in another grf file
158 }
159 }
160 /* The tile has no spritegroup */
161 return 0xFF << 8 | ats->grf_prop.subst_id; // so just give it the substitute
162}

164/* virtual */ uint32_t AirportTileScopeResolver::GetVariable(uint8_t variable, [[maybe_unused]] uint32_t parameter, bool &available) const
165{
166 assert(this->st != nullptr);
167
168 switch (variable) {
169 /* Terrain type */
170 case 0x41: return GetTerrainType(this->tile);
171
172 /* Current town zone of the tile in the nearest town */
173 case 0x42: return to_underlying(GetTownRadiusGroup(ClosestTownFromTile(this->tile, UINT_MAX), this->tile));
174
175 /* Position relative to most northern airport tile. */
176 case 0x43: return GetRelativePosition(this->tile, this->st->airport.tile);
177
178 /* Animation frame of tile */
179 case 0x44: return GetAnimationFrame(this->tile);
180
181 /* Land info of nearby tiles */
182 case 0x60: return GetNearbyAirportTileInformation(parameter, this->tile, this->st->index, this->ro.grffile->grf_version >= 8);
183
184 /* Animation stage of nearby tiles */
185 case 0x61: {
186 TileIndex tile = GetNearbyTile(parameter, this->tile);
187 if (this->st->TileBelongsToAirport(tile)) {
188 return GetAnimationFrame(tile);
189 }
190 return UINT_MAX;
191 }
192
193 /* Get airport tile ID at offset */
194 case 0x62: return GetAirportTileIDAtOffset(GetNearbyTile(parameter, this->tile), this->st, this->ro.grffile->grfid);
195
196 case 0x7A: return GetBadgeVariableResult(*this->ro.grffile, this->ats->badges, parameter);
197 }
198
199 Debug(grf, 1, "Unhandled airport tile variable 0x{:X}", variable);
200
201 available = false;
202 return UINT_MAX;
203}

244static void AirportDrawTileLayout(const TileInfo *ti, const DrawTileSpriteSpan &dts, Colours colour)
245{
246 SpriteID image = dts.ground.sprite;
247 SpriteID pal = dts.ground.pal;

249 if (GB(image, 0, SPRITE_WIDTH) != 0) {

251 DrawWaterClassGround(ti);
252 } else {
253 DrawGroundSprite(image, GroundSpritePaletteTransform(image, pal, GetColourPalette(colour)));

255 }
256
257 DrawNewGRFTileSeq(ti, &dts, TO_BUILDINGS, 0, GetColourPalette(colour));

260bool DrawNewAirportTile(TileInfo *ti, Station *st, const AirportTileSpec *airts)
261{
262 if (ti->tileh != SLOPE_FLAT) {
263 bool draw_old_one = true;
264 if (airts->callback_mask.Test(AirportTileCallbackMask::DrawFoundations)) {
265 /* Called to determine the type (if any) of foundation to draw */
266 uint32_t callback_res = GetAirportTileCallback(CBID_AIRPTILE_DRAW_FOUNDATIONS, 0, 0, airts, st, ti->tile);
267 if (callback_res != CALLBACK_FAILED) draw_old_one = ConvertBooleanCallback(airts->grf_prop.grffile, CBID_AIRPTILE_DRAW_FOUNDATIONS, callback_res);
268 }
269
270 if (draw_old_one) DrawFoundation(ti, FOUNDATION_LEVELED);
271 }
272
273 AirportTileResolverObject object(airts, ti->tile, st);
274 const auto *group = object.Resolve<TileLayoutSpriteGroup>();
275 if (group == nullptr) {
276 return false;
277 }
278
279 auto processor = group->ProcessRegisters(object, nullptr);
280 auto dts = processor.GetLayout();
281 AirportDrawTileLayout(ti, dts, Company::Get(st->owner)->colour);
282 return true;
283}

322bool TriggerAirportAnimation(Station *st, AirportAnimationTrigger trigger, CargoType cargo_type)
323{
324 if (st->airport.tile == INVALID_TILE) return false;
325
326 bool ret = true;
327 uint32_t random = Random();
328 for (TileIndex tile : st->airport) {
329 if (!st->TileBelongsToAirport(tile)) continue;
330
331 const AirportTileSpec *ats = AirportTileSpec::GetByTile(tile);
332 if (ats->grf_prop.grffile == nullptr) continue;
333
334 uint8_t var18_extra = 0;
335 if (IsValidCargoType(cargo_type)) {
336 var18_extra |= ats->grf_prop.grffile->cargo_map[cargo_type] << 8;
337 }
338
339 if (DoTriggerAirportTileAnimation(st, tile, trigger, random, var18_extra)) {
340 SB(random, 0, 16, Random());
341 } else {
342 ret = false;
343 }
344 }
345 return ret;
346}
