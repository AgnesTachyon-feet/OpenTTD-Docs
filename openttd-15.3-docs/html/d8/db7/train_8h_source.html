174 int CalcNextVehicleOffset() const
175 {
176 /* For vehicles with odd lengths the part before the center will be one unit
177 * longer than the part after the center. This means we have to round up the
178 * length of the next vehicle but may not round the length of the current
179 * vehicle. */
180 return this->gcache.cached_veh_length / 2 + (this->Next() != nullptr ? this->Next()->gcache.cached_veh_length + 1 : 0) / 2;
181 }

198 inline uint16_t GetPower() const
199 {
200 /* Power is not added for articulated parts */
201 if (!this->IsArticulatedPart() && HasPowerOnRail(this->railtypes, GetRailType(this->tile))) {
202 uint16_t power = GetVehicleProperty(this, PROP_TRAIN_POWER, RailVehInfo(this->engine_type)->power);
203 /* Halve power for multiheaded parts */
204 if (this->IsMultiheaded()) power /= 2;
205 return power;
206 }
207
208 return 0;
209 }
210
215 inline uint16_t GetPoweredPartPower(const Train *head) const
216 {
217 /* For powered wagons the engine defines the type of engine (i.e. railtype) */
218 if (this->flags.Test(VehicleRailFlag::PoweredWagon) && HasPowerOnRail(head->railtypes, GetRailType(this->tile))) {
219 return RailVehInfo(this->gcache.first_engine)->pow_wag_power;
220 }
221
222 return 0;
223 }
224
229 inline uint16_t GetWeight() const
230 {
231 uint16_t weight = CargoSpec::Get(this->cargo_type)->WeightOfNUnitsInTrain(this->cargo.StoredCount());
232
233 /* Vehicle weight is not added for articulated parts. */
234 if (!this->IsArticulatedPart()) {
235 weight += GetVehicleProperty(this, PROP_TRAIN_WEIGHT, RailVehInfo(this->engine_type)->weight);
236 }
237
238 /* Powered wagons have extra weight added. */
239 if (this->flags.Test(VehicleRailFlag::PoweredWagon)) {
240 weight += RailVehInfo(this->gcache.first_engine)->pow_wag_weight;
241 }
242
243 return weight;
244 }

302 inline uint32_t GetRollingFriction() const
303 {
304 /* Rolling friction for steel on steel is between 0.1% and 0.2%.
305 * The friction coefficient increases with speed in a way that
306 * it doubles at 512 km/h, triples at 1024 km/h and so on. */
307 return 15 * (512 + this->GetCurrentSpeed()) / 512;
308 }
