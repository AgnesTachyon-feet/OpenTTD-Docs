58 inline int ScreenToAnimOffset(const uint32_t *video)
59 {
60 int raw_offset = video - (const uint32_t *)_screen.dst_ptr;
61 if (_screen.pitch == this->anim_buf_pitch) return raw_offset;
62 int lines = raw_offset / _screen.pitch;
63 int across = raw_offset % _screen.pitch;
