22static TileIndex GetBridgeEnd(TileIndex tile, DiagDirection dir)
23{
24 TileIndexDiff delta = TileOffsByDiagDir(dir);
25
26 dir = ReverseDiagDir(dir);

29 } while (!IsBridgeTile(tile) || GetTunnelBridgeDirection(tile) != dir);
