30static uint16_t SanitizeSpriteOffset(uint16_t &num, uint16_t offset, int max_sprites, std::string_view name)
31{
32
33 if (offset >= max_sprites) {
34 GrfMsg(1, "GraphicsNew: {} sprite offset must be less than {}, skipping", name, max_sprites);
35 uint orig_num = num;
36 num = 0;
37 return orig_num;
38 }
39
40 if (offset + num > max_sprites) {
41 GrfMsg(4, "GraphicsNew: {} sprite overflow, truncating...", name);
42 uint orig_num = num;
43 num = std::max(max_sprites - offset, 0);
44 return orig_num - num;
45 }
46
47 return 0;
48}

93/* Action 0x05 */
94static void GraphicsNew(ByteReader &buf)
95{
96 /* <05> <graphics-type> <num-sprites> <other data...>
97 *
98 * B graphics-type What set of graphics the sprites define.
99 * E num-sprites How many sprites are in this set?
100 * V other data Graphics type specific data. Currently unused. */
101
102 uint8_t type = buf.ReadByte();
103 uint16_t num = buf.ReadExtendedByte();
104 uint16_t offset = HasBit(type, 7) ? buf.ReadExtendedByte() : 0;
105 ClrBit(type, 7); // Clear the high bit as that only indicates whether there is an offset.
106
107 if ((type == 0x0D) && (num == 10) && _cur_gps.grfconfig->flags.Test(GRFConfigFlag::System)) {
108 /* Special not-TTDP-compatible case used in openttd.grf
109 * Missing shore sprites and initialisation of SPR_SHORE_BASE */
110 GrfMsg(2, "GraphicsNew: Loading 10 missing shore sprites from extra grf.");
111 LoadNextSprite(SPR_SHORE_BASE + 0, *_cur_gps.file, _cur_gps.nfo_line++); // SLOPE_STEEP_S
112 LoadNextSprite(SPR_SHORE_BASE + 5, *_cur_gps.file, _cur_gps.nfo_line++); // SLOPE_STEEP_W
113 LoadNextSprite(SPR_SHORE_BASE + 7, *_cur_gps.file, _cur_gps.nfo_line++); // SLOPE_WSE
114 LoadNextSprite(SPR_SHORE_BASE + 10, *_cur_gps.file, _cur_gps.nfo_line++); // SLOPE_STEEP_N
115 LoadNextSprite(SPR_SHORE_BASE + 11, *_cur_gps.file, _cur_gps.nfo_line++); // SLOPE_NWS
116 LoadNextSprite(SPR_SHORE_BASE + 13, *_cur_gps.file, _cur_gps.nfo_line++); // SLOPE_ENW
117 LoadNextSprite(SPR_SHORE_BASE + 14, *_cur_gps.file, _cur_gps.nfo_line++); // SLOPE_SEN
118 LoadNextSprite(SPR_SHORE_BASE + 15, *_cur_gps.file, _cur_gps.nfo_line++); // SLOPE_STEEP_E
119 LoadNextSprite(SPR_SHORE_BASE + 16, *_cur_gps.file, _cur_gps.nfo_line++); // SLOPE_EW
120 LoadNextSprite(SPR_SHORE_BASE + 17, *_cur_gps.file, _cur_gps.nfo_line++); // SLOPE_NS
121 if (_loaded_newgrf_features.shore == SHORE_REPLACE_NONE) _loaded_newgrf_features.shore = SHORE_REPLACE_ONLY_NEW;
122 return;
123 }
124
125 /* Supported type? */
126 if ((type >= std::size(_action5_types)) || (_action5_types[type].block_type == A5BLOCK_INVALID)) {
127 GrfMsg(2, "GraphicsNew: Custom graphics (type 0x{:02X}) sprite block of length {} (unimplemented, ignoring)", type, num);
128 _cur_gps.skip_sprites = num;
129 return;
130 }
131
132 const Action5Type *action5_type = &_action5_types[type];
133
134 /* Contrary to TTDP we allow always to specify too few sprites as we allow always an offset,
135 * except for the long version of the shore type:
136 * Ignore offset if not allowed */
137 if ((action5_type->block_type != A5BLOCK_ALLOW_OFFSET) && (offset != 0)) {
138 GrfMsg(1, "GraphicsNew: {} (type 0x{:02X}) do not allow an <offset> field. Ignoring offset.", action5_type->name, type);
139 offset = 0;
140 }
141
142 /* Ignore action5 if too few sprites are specified. (for TTDP compatibility)
143 * This does not make sense, if <offset> is allowed */
144 if ((action5_type->block_type == A5BLOCK_FIXED) && (num < action5_type->min_sprites)) {
145 GrfMsg(1, "GraphicsNew: {} (type 0x{:02X}) count must be at least {}. Only {} were specified. Skipping.", action5_type->name, type, action5_type->min_sprites, num);
146 _cur_gps.skip_sprites = num;
147 return;
148 }
149
150 /* Load at most max_sprites sprites. Skip remaining sprites. (for compatibility with TTDP and future extensions) */
151 uint16_t skip_num = SanitizeSpriteOffset(num, offset, action5_type->max_sprites, action5_type->name);
152 SpriteID replace = action5_type->sprite_base + offset;
153
154 /* Load <num> sprites starting from <replace>, then skip <skip_num> sprites. */
155 GrfMsg(2, "GraphicsNew: Replacing sprites {} to {} of {} (type 0x{:02X}) at SpriteID 0x{:04X}", offset, offset + num - 1, action5_type->name, type, replace);
156
157 if (type == 0x0D) _loaded_newgrf_features.shore = SHORE_REPLACE_ACTION_5;
158
159 if (type == 0x0B) {
160 static const SpriteID depot_with_track_offset = SPR_TRAMWAY_DEPOT_WITH_TRACK - SPR_TRAMWAY_BASE;
161 static const SpriteID depot_no_track_offset = SPR_TRAMWAY_DEPOT_NO_TRACK - SPR_TRAMWAY_BASE;
162 if (offset <= depot_with_track_offset && offset + num > depot_with_track_offset) _loaded_newgrf_features.tram = TRAMWAY_REPLACE_DEPOT_WITH_TRACK;
163 if (offset <= depot_no_track_offset && offset + num > depot_no_track_offset) _loaded_newgrf_features.tram = TRAMWAY_REPLACE_DEPOT_NO_TRACK;
164 }
165
166 /* If the baseset or grf only provides sprites for flat tiles (pre #10282), duplicate those for use on slopes. */
167 bool dup_oneway_sprites = ((type == 0x09) && (offset + num <= ONEWAY_SLOPE_N_OFFSET));
168
169 for (; num > 0; num--) {
170 _cur_gps.nfo_line++;
171 SpriteID load_index = (replace == 0 ? _cur_gps.spriteid++ : replace++);
172 LoadNextSprite(load_index, *_cur_gps.file, _cur_gps.nfo_line);
173 if (dup_oneway_sprites) {
174 DupSprite(load_index, load_index + ONEWAY_SLOPE_N_OFFSET);
175 DupSprite(load_index, load_index + ONEWAY_SLOPE_S_OFFSET);
176 }
177 }
178
179 _cur_gps.skip_sprites = skip_num;
180}

182/* Action 0x05 (SKIP) */
183static void SkipAct5(ByteReader &buf)
184{
185 /* Ignore type byte */
186 buf.ReadByte();
187
188 /* Skip the sprites of this action */
189 _cur_gps.skip_sprites = buf.ReadExtendedByte();
190
191 GrfMsg(3, "SkipAct5: Skipping {} sprites", _cur_gps.skip_sprites);
192}
