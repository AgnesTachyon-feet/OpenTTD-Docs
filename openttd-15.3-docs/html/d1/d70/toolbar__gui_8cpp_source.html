136static void PopupMainToolbarMenu(Window *w, WidgetID widget, const std::initializer_list<StringID> &strings)
137{
138 DropDownList list;
139 int i = 0;
140 for (StringID string : strings) {
141 if (string == STR_NULL) {
142 list.push_back(MakeDropDownListDividerItem());
143 } else {
144 list.push_back(MakeDropDownListStringItem(string, i));
145 i++;
146 }
147 }
148 PopupMainToolbarMenu(w, widget, std::move(list), 0);
149}

162static void PopupMainCompanyToolbMenu(Window *w, WidgetID widget, CompanyMask grey = {})
163{
164 DropDownList list;
165
166 switch (widget) {
167 case WID_TN_COMPANIES:
168 if (!_networking) break;
169
170 /* Add the client list button for the companies menu */
171 list.push_back(MakeDropDownListStringItem(STR_NETWORK_COMPANY_LIST_CLIENT_LIST, CTMN_CLIENT_LIST));
172
173 if (_local_company != COMPANY_SPECTATOR) {
174 list.push_back(MakeDropDownListStringItem(STR_NETWORK_COMPANY_LIST_SPECTATE, CTMN_SPECTATE));
175 }
176 break;
177 case WID_TN_STORY:
178 list.push_back(MakeDropDownListStringItem(STR_STORY_BOOK_SPECTATOR, CTMN_SPECTATOR));
179 break;
180
181 case WID_TN_GOAL:
182 list.push_back(MakeDropDownListStringItem(STR_GOALS_SPECTATOR, CTMN_SPECTATOR));
183 break;
184 }
185
186 for (CompanyID c = CompanyID::Begin(); c < MAX_COMPANIES; ++c) {
187 if (!Company::IsValidID(c)) continue;
188 list.push_back(std::make_unique<DropDownListCompanyItem>(c, grey.Test(c)));
189 }
190
191 PopupMainToolbarMenu(w, widget, std::move(list), _local_company == COMPANY_SPECTATOR ? (widget == WID_TN_COMPANIES ? CTMN_CLIENT_LIST : CTMN_SPECTATOR) : _local_company.base());
192}

310static CallBackFunction MenuClickSettings(int index)
311{
312 switch (index) {
313 case OME_GAMEOPTIONS: ShowGameOptions(); return CBF_NONE;
314 case OME_AI_SETTINGS: ShowAIConfigWindow(); return CBF_NONE;
315 case OME_GAMESCRIPT_SETTINGS: ShowGSConfigWindow(); return CBF_NONE;

317 case OME_SANDBOX: ShowCheatWindow(); break;
318 case OME_TRANSPARENCIES: ShowTransparencyToolbar(); break;
319
320 case OME_SHOW_TOWNNAMES: ToggleBit(_display_opt, DO_SHOW_TOWN_NAMES); break;
321 case OME_SHOW_STATIONNAMES: ToggleBit(_display_opt, DO_SHOW_STATION_NAMES); break;
322 case OME_SHOW_STATIONNAMES_TRAIN: _facility_display_opt.Flip(StationFacility::Train); break;
323 case OME_SHOW_STATIONNAMES_LORRY: _facility_display_opt.Flip(StationFacility::TruckStop); break;
324 case OME_SHOW_STATIONNAMES_BUS: _facility_display_opt.Flip(StationFacility::BusStop); break;
325 case OME_SHOW_STATIONNAMES_SHIP: _facility_display_opt.Flip(StationFacility::Dock); break;
326 case OME_SHOW_STATIONNAMES_PLANE: _facility_display_opt.Flip(StationFacility::Airport); break;
327 case OME_SHOW_STATIONNAMES_GHOST: _facility_display_opt.Flip(STATION_FACILITY_GHOST); break;
328 case OME_SHOW_WAYPOINTNAMES: ToggleBit(_display_opt, DO_SHOW_WAYPOINT_NAMES); break;
329 case OME_SHOW_SIGNS: ToggleBit(_display_opt, DO_SHOW_SIGNS); break;
330 case OME_SHOW_COMPETITOR_SIGNS:
331 ToggleBit(_display_opt, DO_SHOW_COMPETITOR_SIGNS);
332 InvalidateWindowClassesData(WC_SIGN_LIST, -1);
333 break;
334 case OME_FULL_ANIMATION: ToggleBit(_display_opt, DO_FULL_ANIMATION); CheckBlitter(); break;
335 case OME_FULL_DETAILS: ToggleBit(_display_opt, DO_FULL_DETAIL); break;
336 case OME_TRANSPARENTBUILDINGS: ToggleTransparency(TO_HOUSES); break;
337 case OME_SHOW_STATIONSIGNS: ToggleTransparency(TO_SIGNS); break;
338 }
339 MarkWholeScreenDirty();
340 return CBF_NONE;
341}

1133void SetStartingYear(TimerGameCalendar::Year year)
1134{
1135 _settings_game.game_creation.starting_year = Clamp(year, CalendarTime::MIN_YEAR, CalendarTime::MAX_YEAR);
1136 TimerGameCalendar::Date new_calendar_date = TimerGameCalendar::ConvertYMDToDate(_settings_game.game_creation.starting_year, 0, 1);
1137 TimerGameEconomy::Date new_economy_date{new_calendar_date.base()};
1138
1139 /* We must set both Calendar and Economy dates to keep them in sync. Calendar first. */
1140 TimerGameCalendar::SetDate(new_calendar_date, 0);
1141
1142 /* If you open a savegame as a scenario, there may already be link graphs and/or vehicles. These use economy date. */
1143 LinkGraphSchedule::instance.ShiftDates(new_economy_date - TimerGameEconomy::date);
1144 for (auto v : Vehicle::Iterate()) v->ShiftDates(new_economy_date - TimerGameEconomy::date);
1145
1146 /* Only change the date after changing cached values above. */
1147 TimerGameEconomy::SetDate(new_economy_date, 0);
1148}

1374 void SetupSmallestSize(Window *w) override
1375 {
1376 this->smallest_x = 0; // Biggest child
1377 this->smallest_y = 0; // Biggest child
1378 this->fill_x = 1;
1379 this->fill_y = 0;
1380 this->resize_x = 1; // We only resize in this direction
1381 this->resize_y = 0; // We never resize in this direction
1382 this->spacers = 0;
1383
1384 uint nbuttons = 0;
1385 /* First initialise some variables... */
1386 for (const auto &child_wid : this->children) {
1387 child_wid->SetupSmallestSize(w);
1388 this->smallest_y = std::max(this->smallest_y, child_wid->smallest_y + child_wid->padding.Vertical());
1389 if (this->IsButton(child_wid->type)) {
1390 nbuttons++;
1391 this->smallest_x = std::max(this->smallest_x, child_wid->smallest_x + child_wid->padding.Horizontal());
1392 } else if (child_wid->type == NWID_SPACER) {
1393 this->spacers++;
1394 }
1395 }
1396
1397 /* ... then in a second pass make sure the 'current' heights are set. Won't change ever. */
1398 for (const auto &child_wid : this->children) {
1399 child_wid->current_y = this->smallest_y;
1400 if (!this->IsButton(child_wid->type)) {
1401 child_wid->current_x = child_wid->smallest_x;
1402 }
1403 }
1404
1405 /* Exclude the switcher button which is not displayed when the toolbar fits the screen. When the switch is
1406 * displayed there will be no spacers anyway. */
1407 --nbuttons;
1408
1409 /* Allow space for all buttons, and include spacers at quarter the width of buttons. */
1410 _toolbar_width = nbuttons * this->smallest_x + this->spacers * this->smallest_x / 4;
1411 }
1412
1413 void AssignSizePosition(SizingType sizing, int x, int y, uint given_width, uint given_height, bool rtl) override
1414 {
1415 assert(given_width >= this->smallest_x && given_height >= this->smallest_y);
1416
1417 this->pos_x = x;
1418 this->pos_y = y;
1419 this->current_x = given_width;
1420 this->current_y = given_height;
1421
1422 /* Figure out what are the visible buttons */
1423 uint arrangeable_count, button_count, spacer_count;
1424 const WidgetID *arrangement = GetButtonArrangement(given_width, arrangeable_count, button_count, spacer_count);
1425
1426 /* Create us ourselves a quick lookup table from WidgetID to slot. */
1427 std::map<WidgetID, uint> lookup;
1428 for (auto it = std::begin(this->children); it != std::end(this->children); ++it) {
1429 NWidgetBase *nwid = it->get();
1430 nwid->current_x = 0; /* Hide widget, it will be revealed in the next step. */
1431 if (nwid->type == NWID_SPACER) continue;
1432 NWidgetCore *nwc = dynamic_cast<NWidgetCore *>(nwid);
1433 assert(nwc != nullptr);
1434 lookup[nwc->GetIndex()] = std::distance(this->children.begin(), it);
1435 }
1436
1437 /* Now assign the widgets to their rightful place */
1438 uint position = 0; // Place to put next child relative to origin of the container.
1439 uint spacer_space = std::max(0, (int)given_width - (int)(button_count * this->smallest_x)); // Remaining spacing for 'spacer' widgets
1440 uint button_space = given_width - spacer_space; // Remaining spacing for the buttons
1441 uint spacer_i = 0;
1442 uint button_i = 0;
1443
1444 /* Index into the arrangement indices. */
1445 const WidgetID *slotp = rtl ? &arrangement[arrangeable_count - 1] : arrangement;
1446 for (uint i = 0; i < arrangeable_count; i++) {
1447 uint slot = lookup[*slotp];
1448 auto &child_wid = this->children[slot];
1449 /* If we have space to give to the spacers, do that. */
1450 if (spacer_space > 0 && slot > 0 && slot < this->children.size() - 1) {
1451 const auto &possible_spacer = this->children[slot + (rtl ? 1 : -1)];
1452 if (possible_spacer != nullptr && possible_spacer->type == NWID_SPACER) {
1453 uint add = spacer_space / (spacer_count - spacer_i);
1454 position += add;
1455 spacer_space -= add;
1456 spacer_i++;
1457 }
1458 }
1459
1460 /* Buttons can be scaled, the others not. */
1461 if (this->IsButton(child_wid->type)) {
1462 child_wid->current_x = button_space / (button_count - button_i);
1463 button_space -= child_wid->current_x;
1464 button_i++;
1465 } else {
1466 child_wid->current_x = child_wid->smallest_x;
1467 }
1468 child_wid->AssignSizePosition(sizing, x + position, y, child_wid->current_x, this->current_y, rtl);
1469 position += child_wid->current_x;
1470
1471 if (rtl) {
1472 slotp--;
1473 } else {
1474 slotp++;
1475 }
1476 }
1477 }
1478
1479 void Draw(const Window *w) override
1480 {
1481 /* Draw brown-red toolbar bg. */
1482 const Rect r = this->GetCurrentRect();
1483 GfxFillRect(r, PC_VERY_DARK_RED);
1484 GfxFillRect(r, PC_DARK_RED, FILLRECT_CHECKER);
1485
1486 this->NWidgetContainer::Draw(w);
1487 }

1497 virtual const WidgetID *GetButtonArrangement(uint &width, uint &arrangeable_count, uint &button_count, uint &spacer_count) const = 0;
1498};
