74inline void MakeObject(Tile t, Owner o, ObjectID index, WaterClass wc, uint8_t random)
75{
76 SetTileType(t, MP_OBJECT);
77 SetTileOwner(t, o);
78 SetWaterClass(t, wc);
79 t.m2() = index.base();
80 t.m3() = random;
81 t.m4() = 0;
82 t.m5() = index.base() >> 16;
