54uint32_t GetRelativePosition(TileIndex tile, TileIndex ind_tile)
55{
56 uint8_t x = TileX(tile) - TileX(ind_tile);
57 uint8_t y = TileY(tile) - TileY(ind_tile);
58
59 return ((y & 0xF) << 20) | ((x & 0xF) << 16) | (y << 8) | x;
60}

62/* virtual */ uint32_t IndustryTileScopeResolver::GetVariable(uint8_t variable, [[maybe_unused]] uint32_t parameter, bool &available) const
63{
64 switch (variable) {
65 /* Construction state of the tile: a value between 0 and 3 */
66 case 0x40: return (IsTileType(this->tile, MP_INDUSTRY)) ? GetIndustryConstructionStage(this->tile) : 0;
67
68 /* Terrain type */
69 case 0x41: return GetTerrainType(this->tile);
70
71 /* Current town zone of the tile in the nearest town */
72 case 0x42: return to_underlying(GetTownRadiusGroup(ClosestTownFromTile(this->tile, UINT_MAX), this->tile));
73
74 /* Relative position */
75 case 0x43: return GetRelativePosition(this->tile, this->industry->location.tile);
76
77 /* Animation frame. Like house variable 46 but can contain anything 0..FF. */
78 case 0x44: return IsTileType(this->tile, MP_INDUSTRY) ? GetAnimationFrame(this->tile) : 0;
79
80 /* Land info of nearby tiles */
81 case 0x60: return GetNearbyIndustryTileInformation(parameter, this->tile,
82 this->industry == nullptr ? (IndustryID)IndustryID::Invalid() : this->industry->index, true, this->ro.grffile->grf_version >= 8);
83
84 /* Animation stage of nearby tiles */
85 case 0x61: {
86 TileIndex tile = GetNearbyTile(parameter, this->tile);
87 if (IsTileType(tile, MP_INDUSTRY) && Industry::GetByTile(tile) == this->industry) {
88 return GetAnimationFrame(tile);
89 }
90 return UINT_MAX;
91 }
92
93 /* Get industry tile ID at offset */
94 case 0x62: return GetIndustryIDAtOffset(GetNearbyTile(parameter, this->tile), this->industry, this->ro.grffile->grfid);
95
96 case 0x7A: return GetBadgeVariableResult(*this->ro.grffile, GetIndustryTileSpec(GetIndustryGfx(this->tile))->badges, parameter);
97 }
98
99 Debug(grf, 1, "Unhandled industry tile variable 0x{:X}", variable);
100
101 available = false;
102 return UINT_MAX;
103}

161static void IndustryDrawTileLayout(const TileInfo *ti, const DrawTileSpriteSpan &dts, Colours rnd_colour, uint8_t stage)
162{
163 SpriteID image = dts.ground.sprite;
164 PaletteID pal = dts.ground.pal;
165
166 if (HasBit(image, SPRITE_MODIFIER_CUSTOM_SPRITE)) image += stage;
167 if (HasBit(pal, SPRITE_MODIFIER_CUSTOM_SPRITE)) pal += stage;
168
169 if (GB(image, 0, SPRITE_WIDTH) != 0) {
170 /* If the ground sprite is the default flat water sprite, draw also canal/river borders
171 * Do not do this if the tile's WaterClass is 'land'. */
172 if (image == SPR_FLAT_WATER_TILE && IsTileOnWater(ti->tile)) {
173 DrawWaterClassGround(ti);
174 } else {
175 DrawGroundSprite(image, GroundSpritePaletteTransform(image, pal, GetColourPalette(rnd_colour)));
176 }
177 }
178
179 DrawNewGRFTileSeq(ti, &dts, TO_INDUSTRIES, stage, GetColourPalette(rnd_colour));
180}

191bool DrawNewIndustryTile(TileInfo *ti, Industry *i, IndustryGfx gfx, const IndustryTileSpec *inds)
192{
193 if (ti->tileh != SLOPE_FLAT) {
194 bool draw_old_one = true;
195 if (inds->callback_mask.Test(IndustryTileCallbackMask::DrawFoundations)) {
196 /* Called to determine the type (if any) of foundation to draw for industry tile */
197 uint32_t callback_res = GetIndustryTileCallback(CBID_INDTILE_DRAW_FOUNDATIONS, 0, 0, gfx, i, ti->tile);
198 if (callback_res != CALLBACK_FAILED) draw_old_one = ConvertBooleanCallback(inds->grf_prop.grffile, CBID_INDTILE_DRAW_FOUNDATIONS, callback_res);
199 }
200
201 if (draw_old_one) DrawFoundation(ti, FOUNDATION_LEVELED);
202 }
203
204 IndustryTileResolverObject object(gfx, ti->tile, i);
205
206 const auto *group = object.Resolve<TileLayoutSpriteGroup>();
207 if (group == nullptr) return false;
208
209 /* Limit the building stage to the number of stages supplied. */
210 uint8_t stage = GetIndustryConstructionStage(ti->tile);
211 auto processor = group->ProcessRegisters(object, &stage);
212 auto dts = processor.GetLayout();
213 IndustryDrawTileLayout(ti, dts, i->random_colour, stage);
214 return true;
215}

232CommandCost PerformIndustryTileSlopeCheck(TileIndex ind_base_tile, TileIndex ind_tile, const IndustryTileSpec *its, IndustryType type, IndustryGfx gfx, size_t layout_index, uint16_t initial_random_bits, Owner founder, IndustryAvailabilityCallType creation_type)
233{
234 Industry ind;
235 ind.index = IndustryID::Invalid();
236 ind.location.tile = ind_base_tile;
237 ind.location.w = 0;
238 ind.type = type;
239 ind.random = initial_random_bits;
240 ind.founder = founder;
241
242 std::array<int32_t, 16> regs100;
243 uint16_t callback_res = GetIndustryTileCallback(CBID_INDTILE_SHAPE_CHECK, 0, creation_type << 8 | static_cast<uint32_t>(layout_index), gfx, &ind, ind_tile, regs100);
244 if (callback_res == CALLBACK_FAILED) {
245 if (!IsSlopeRefused(GetTileSlope(ind_tile), its->slopes_refused)) return CommandCost();
246 return CommandCost(STR_ERROR_SITE_UNSUITABLE);
247 }
248 if (its->grf_prop.grffile->grf_version < 7) {
249 if (callback_res != 0) return CommandCost();
250 return CommandCost(STR_ERROR_SITE_UNSUITABLE);
251 }
252
253 return GetErrorMessageFromLocationCallbackResult(callback_res, regs100, its->grf_prop.grffile, STR_ERROR_SITE_UNSUITABLE);
254}

324static void DoTriggerIndustryTileRandomisation(TileIndex tile, IndustryRandomTrigger trigger, Industry *ind, uint32_t &reseed_industry)
325{
326 assert(IsValidTile(tile) && IsTileType(tile, MP_INDUSTRY));
327
328 IndustryGfx gfx = GetIndustryGfx(tile);
329 const IndustryTileSpec *itspec = GetIndustryTileSpec(gfx);
330
331 if (!itspec->grf_prop.HasSpriteGroups()) return;
332
333 IndustryTileResolverObject object(gfx, tile, ind, CBID_RANDOM_TRIGGER);
334 auto waiting_random_triggers = GetIndustryRandomTriggers(tile);
335 waiting_random_triggers.Set(trigger);
336 SetIndustryRandomTriggers(tile, waiting_random_triggers); // store now for var 5F
337 object.SetWaitingRandomTriggers(waiting_random_triggers);
338
339 object.ResolveRerandomisation();
340
341 /* Store remaining triggers. */
342 waiting_random_triggers.Reset(object.GetUsedRandomTriggers());
343 SetIndustryRandomTriggers(tile, waiting_random_triggers);
344
345 /* Rerandomise tile bits */
346 uint8_t new_random_bits = Random();
347 uint8_t random_bits = GetIndustryRandomBits(tile);
348 random_bits &= ~object.reseed[VSG_SCOPE_SELF];
349 random_bits |= new_random_bits & object.reseed[VSG_SCOPE_SELF];
350 SetIndustryRandomBits(tile, random_bits);
351 MarkTileDirtyByTile(tile);
352
353 reseed_industry |= object.reseed[VSG_SCOPE_PARENT];
354}

361static void DoReseedIndustry(Industry *ind, uint32_t reseed)
362{
363 if (reseed == 0 || ind == nullptr) return;
364
365 uint16_t random_bits = Random();
366 ind->random &= reseed;
367 ind->random |= random_bits & reseed;
