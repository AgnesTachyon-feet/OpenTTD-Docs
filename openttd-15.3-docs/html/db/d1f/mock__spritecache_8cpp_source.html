19static bool MockLoadNextSprite(SpriteID load_index)
20{
21 UniquePtrSpriteAllocator allocator;
22 allocator.Allocate<Sprite>(sizeof(Sprite));
23
24 bool is_mapgen = IsMapgenSpriteID(load_index);
25
26 SpriteCache *sc = AllocateSpriteCache(load_index);
27 sc->file = nullptr;
28 sc->file_pos = 0;
29 sc->ptr = std::move(allocator.data);
30 sc->length = static_cast<uint32_t>(allocator.size);
31 sc->lru = 0;
32 sc->id = 0;
33 sc->type = is_mapgen ? SpriteType::MapGen : SpriteType::Normal;
34 sc->warned = false;
35 sc->control_flags = {};
36
37 /* Fill with empty sprites up until the default sprite count. */
38 return load_index < SPR_OPENTTD_BASE + OPENTTD_SPRITE_COUNT;
39}
