24enum PacketGameType : uint8_t {
25 /*
26 * These first ten packets must remain in this order for backward and forward compatibility
27 * between clients that are trying to join directly. These packets can be received and/or sent
28 * by the server before the server has processed the 'join' packet from the client.
29 */
30
31 /* Packets sent by socket accepting code without ever constructing a client socket instance. */
32 PACKET_SERVER_FULL,
33 PACKET_SERVER_BANNED,
34
35 /* Packets used by the client to join and an error message when the revision is wrong. */
36 PACKET_CLIENT_JOIN,
37 PACKET_SERVER_ERROR,
38
39 /* Unused packet types, formerly used for the pre-game lobby. */
40 PACKET_CLIENT_UNUSED,
41 PACKET_SERVER_UNUSED,
42
43 /* Packets used to get the game info. */
44 PACKET_SERVER_GAME_INFO,
45 PACKET_CLIENT_GAME_INFO,
46
47 /* A server quitting this game. */
48 PACKET_SERVER_NEWGAME,
49 PACKET_SERVER_SHUTDOWN,
50
51 /*
52 * Packets after here assume that the client
53 * and server are running the same version. As
54 * such ordering is unimportant from here on.
55 *
56 * The following is the remainder of the packets
57 * sent as part of authenticating and getting
58 * the map and other important data.
59 */
60
61 /* After the join step, the first perform game authentication and enabling encryption. */
62 PACKET_SERVER_AUTH_REQUEST,
63 PACKET_CLIENT_AUTH_RESPONSE,
64 PACKET_SERVER_ENABLE_ENCRYPTION,
65
66 /* After the authentication is done, the next step is identification. */
67 PACKET_CLIENT_IDENTIFY,
68
69 /* After the identify step, the next is checking NewGRFs. */
70 PACKET_SERVER_CHECK_NEWGRFS,
71 PACKET_CLIENT_NEWGRFS_CHECKED,
72
73 /* The server welcomes the authenticated client and sends information of other clients. */
74 PACKET_SERVER_WELCOME,
75 PACKET_SERVER_CLIENT_INFO,
76
77 /* Getting the savegame/map. */
78 PACKET_CLIENT_GETMAP,
79 PACKET_SERVER_WAIT,
80 PACKET_SERVER_MAP_BEGIN,
81 PACKET_SERVER_MAP_SIZE,
82 PACKET_SERVER_MAP_DATA,
83 PACKET_SERVER_MAP_DONE,
84 PACKET_CLIENT_MAP_OK,
85
86 PACKET_SERVER_JOIN,
87
88 /*
89 * At this moment the client has the map and
90 * the client is fully authenticated. Now the
91 * normal communication starts.
92 */
93
94 /* Game progress monitoring. */
95 PACKET_SERVER_FRAME,
96 PACKET_CLIENT_ACK,
97 PACKET_SERVER_SYNC,
98
99 /* Sending commands around. */
100 PACKET_CLIENT_COMMAND,
101 PACKET_SERVER_COMMAND,
102
103 /* Human communication! */
104 PACKET_CLIENT_CHAT,
105 PACKET_SERVER_CHAT,
106 PACKET_SERVER_EXTERNAL_CHAT,
107
108 /* Remote console. */
109 PACKET_CLIENT_RCON,
110 PACKET_SERVER_RCON,
111
112 /* Moving a client.*/
113 PACKET_CLIENT_MOVE,
114 PACKET_SERVER_MOVE,
115
116 /* Configuration updates. */
117 PACKET_CLIENT_SET_NAME,
118 PACKET_SERVER_CONFIG_UPDATE,
119
120 /* A client quitting. */
121 PACKET_CLIENT_QUIT,
122 PACKET_SERVER_QUIT,
123 PACKET_CLIENT_ERROR,
124 PACKET_SERVER_ERROR_QUIT,
125
126 PACKET_END,
127};
