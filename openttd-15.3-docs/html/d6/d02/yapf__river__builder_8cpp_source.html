85 inline void PfFollowNode(Node &old_node)
86 {
87 for (DiagDirection d = DIAGDIR_BEGIN; d < DIAGDIR_END; ++d) {
88 const TileIndex t = old_node.GetTile() + TileOffsByDiagDir(d);
89 if (IsValidTile(t) && RiverFlowsDown(old_node.GetTile(), t)) {
90 Node &node = Yapf().CreateNewNode();
91 node.Set(&old_node, t, INVALID_TRACKDIR, true);

102 static void BuildRiver(TileIndex start_tile, TileIndex end_tile, TileIndex spring_tile, bool main_river)
103 {
104 YapfRiverBuilder pf(start_tile, end_tile);
105 if (pf.FindPath(nullptr) == false) return; // No path found
106
107 /* First, build the river without worrying about its width. */
108 for (Node *node = pf.GetBestNode(); node != nullptr; node = node->parent) {
109 TileIndex tile = node->GetTile();
110 if (!IsWaterTile(tile)) {
111 MakeRiverAndModifyDesertZoneAround(tile);
112 }
113 }
114
115 /* If the river is a main river, go back along the path to widen it.
116 * Don't make wide rivers if we're using the original landscape generator.
117 */
118 if (_settings_game.game_creation.land_generator != LG_ORIGINAL && main_river) {
119 const uint long_river_length = _settings_game.game_creation.min_river_length * 4;
120
121 for (Node *node = pf.GetBestNode(); node != nullptr; node = node->parent) {
122 const TileIndex center_tile = node->GetTile();
123
124 /* Check if we should widen river depending on how far we are away from the source. */
125 uint current_river_length = DistanceManhattan(spring_tile, center_tile);
126 uint diameter = std::min(3u, (current_river_length / (long_river_length / 3u)) + 1u);
127 if (diameter <= 1) continue;
128
129 for (auto tile : SpiralTileSequence(center_tile, diameter)) {
130 RiverMakeWider(tile, center_tile);
131 }
132 }
133 }
134 }
135};
