21bool MayHaveRoad(Tile t)
22{
23 switch (GetTileType(t)) {
24 case MP_ROAD:
25 return true;
26
27 case MP_STATION:
28 return IsAnyRoadStop(t);
29
30 case MP_TUNNELBRIDGE:

54RoadBits GetAnyRoadBits(Tile tile, RoadTramType rtt, bool straight_tunnel_bridge_entrance)
55{
56 if (!MayHaveRoad(tile) || !HasTileRoadType(tile, rtt)) return ROAD_NONE;
57
58 switch (GetTileType(tile)) {
59 case MP_ROAD:
60 switch (GetRoadTileType(tile)) {
61 default:
62 case RoadTileType::Normal: return GetRoadBits(tile, rtt);
63 case RoadTileType::Crossing: return GetCrossingRoadBits(tile);
64 case RoadTileType::Depot: return DiagDirToRoadBits(GetRoadDepotDirection(tile));
65 }
66
67 case MP_STATION:
68 assert(IsAnyRoadStopTile(tile)); // ensured by MayHaveRoad
69 if (IsDriveThroughStopTile(tile)) return AxisToRoadBits(GetDriveThroughStopAxis(tile));
70 return DiagDirToRoadBits(GetBayRoadStopDir(tile));
71
72 case MP_TUNNELBRIDGE:
73 assert(GetTunnelBridgeTransportType(tile) == TRANSPORT_ROAD); // ensured by MayHaveRoad
74 return straight_tunnel_bridge_entrance ?
75 AxisToRoadBits(DiagDirToAxis(GetTunnelBridgeDirection(tile))) :
76 DiagDirToRoadBits(ReverseDiagDir(GetTunnelBridgeDirection(tile)));
77
78 default: return ROAD_NONE;
79 }
80}
