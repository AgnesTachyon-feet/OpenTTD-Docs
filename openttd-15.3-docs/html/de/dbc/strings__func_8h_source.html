22inline StringTab GetStringTab(StringID str)
23{
24 StringTab result = (StringTab)(str >> TAB_SIZE_BITS);
25 if (result >= TEXT_TAB_NEWGRF_START) return TEXT_TAB_NEWGRF_START;
26 if (result >= TEXT_TAB_GAMESCRIPT_START) return TEXT_TAB_GAMESCRIPT_START;
27 return result;
28}

46inline StringID MakeStringID(StringTab tab, StringIndexInTab index)
47{
48 if (tab == TEXT_TAB_NEWGRF_START) {
49 assert(index < TAB_SIZE_NEWGRF);
50 } else if (tab == TEXT_TAB_GAMESCRIPT_START) {
51 assert(index < TAB_SIZE_GAMESCRIPT);
52 } else {
53 assert(tab < TEXT_TAB_END);
54 assert(index < TAB_SIZE);
55 }
56 return (tab << TAB_SIZE_BITS) + index.base();
57}

83inline int64_t PackVelocity(uint speed, VehicleType type)
84{
85 /* Vehicle type is a byte, so packed into the top 8 bits of the 64-bit
86 * parameter, although only values from 0-3 are relevant. */
87 return speed | (static_cast<uint64_t>(type) << 56);
