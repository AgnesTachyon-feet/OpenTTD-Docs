183inline RoadTypes GetMaskForRoadTramType(RoadTramType rtt)
184{
185 extern RoadTypes _roadtypes_road;
186 extern RoadTypes _roadtypes_tram;
187 return rtt == RTT_ROAD ? _roadtypes_road : _roadtypes_tram;
188}

190inline bool RoadTypeIsRoad(RoadType roadtype)
191{
192 return GetMaskForRoadTramType(RTT_ROAD).Test(roadtype);
193}

195inline bool RoadTypeIsTram(RoadType roadtype)

240inline Money RoadBuildCost(RoadType roadtype)
241{
242 assert(roadtype < ROADTYPE_END);
243 return (_price[PR_BUILD_ROAD] * GetRoadTypeInfo(roadtype)->cost_multiplier) >> 3;
244}
245
251inline Money RoadClearCost(RoadType roadtype)
252{
253 assert(roadtype < ROADTYPE_END);
254
255 /* Flat fee for removing road. */
256 if (RoadTypeIsRoad(roadtype)) return _price[PR_CLEAR_ROAD];
257
258 /* Clearing tram earns a little money, but also incurs the standard clear road cost,
259 * so no profit can be made. */
260 return _price[PR_CLEAR_ROAD] - RoadBuildCost(roadtype) * 3 / 4;
261}
262
269inline Money RoadConvertCost(RoadType from, RoadType to)
270{
271 /* Don't apply convert costs when converting to the same roadtype (ex. building a roadstop over existing road) */
272 if (from == to) return (Money)0;
273
274 /* Same cost as removing and then building. */
275 return RoadBuildCost(to) + RoadClearCost(from);
276}

283inline bool RoadNoLevelCrossing(RoadType roadtype)
284{
285 assert(roadtype < ROADTYPE_END);
286 return GetRoadTypeInfo(roadtype)->flags.Test(RoadTypeFlag::NoLevelCrossing);
287}
