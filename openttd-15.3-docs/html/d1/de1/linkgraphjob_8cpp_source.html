86LinkGraphJob::~LinkGraphJob()
87{
88 this->JoinThread();
89
90 /* Don't update stuff from other pools, when everything is being removed.
91 * Accessing other pools may be invalid. */
92 if (CleaningPool()) return;
93
94 /* If the job has been aborted, the job state is invalid.
95 * This should never be reached, as once the job has been marked as aborted
96 * the only valid job operation is to clear the LinkGraphJob pool. */
97 assert(!this->IsJobAborted());
98
99 /* Link graph has been merged into another one. */
100 if (!LinkGraph::IsValidID(this->link_graph.index)) return;
101
102 uint16_t size = this->Size();
103 for (NodeID node_id = 0; node_id < size; ++node_id) {
104 NodeAnnotation &from = this->nodes[node_id];
105
106 /* The station can have been deleted. Remove all flows originating from it then. */
107 Station *st = Station::GetIfValid(from.base.station);
108 if (st == nullptr) {
109 this->EraseFlows(from.base.station);
110 continue;
111 }
112
113 /* Link graph merging and station deletion may change around IDs. Make
114 * sure that everything is still consistent or ignore it otherwise. */
115 GoodsEntry &ge = st->goods[this->Cargo()];
116 if (ge.link_graph != this->link_graph.index || ge.node != node_id) {
117 this->EraseFlows(from.base.station);
118 continue;
119 }
120
121 LinkGraph *lg = LinkGraph::Get(ge.link_graph);
122 FlowStatMap &flows = from.flows;
123 FlowStatMap &geflows = ge.GetOrCreateData().flows;
124
125 for (const auto &edge : from.edges) {
126 if (edge.Flow() == 0) continue;
127 NodeID dest_id = edge.base.dest_node;
128 StationID to = this->nodes[dest_id].base.station;
129 Station *st2 = Station::GetIfValid(to);
130 if (st2 == nullptr || st2->goods[this->Cargo()].link_graph != this->link_graph.index ||
131 st2->goods[this->Cargo()].node != dest_id ||
132 !(*lg)[node_id].HasEdgeTo(dest_id) ||
133 (*lg)[node_id][dest_id].LastUpdate() == EconomyTime::INVALID_DATE) {
134 /* Edge has been removed. Delete flows. */
135 std::vector<StationID> erased = flows.DeleteFlows(to);
136 /* Delete old flows for source stations which have been deleted
137 * from the new flows. This avoids flow cycles between old and
138 * new flows. */
139 for (const StationID &station : erased) geflows.erase(station);
140 } else if ((*lg)[node_id][dest_id].last_unrestricted_update == EconomyTime::INVALID_DATE) {
141 /* Edge is fully restricted. */
142 flows.RestrictFlows(to);
143 }
144 }
145
146 /* Swap shares and invalidate ones that are completely deleted. Don't
147 * really delete them as we could then end up with unroutable cargo
148 * somewhere. Do delete them and also reroute relevant cargo if
149 * automatic distribution has been turned off for that cargo. */
150 for (FlowStatMap::iterator it(geflows.begin()); it != geflows.end();) {
151 FlowStatMap::iterator new_it = flows.find(it->first);
152 if (new_it == flows.end()) {
153 if (_settings_game.linkgraph.GetDistributionType(this->Cargo()) != DT_MANUAL) {
154 it->second.Invalidate();
155 ++it;
156 } else {
157 FlowStat shares(StationID::Invalid(), 1);
158 it->second.SwapShares(shares);
159 geflows.erase(it++);
160 for (FlowStat::SharesMap::const_iterator shares_it(shares.GetShares()->begin());
161 shares_it != shares.GetShares()->end(); ++shares_it) {
162 RerouteCargo(st, this->Cargo(), shares_it->second, st->index);
163 }
164 }
165 } else {
166 it->second.SwapShares(new_it->second);
167 flows.erase(new_it);
168 ++it;
169 }
170 }
171 geflows.insert(flows.begin(), flows.end());
172 if (ge.GetData().IsEmpty()) ge.ClearData();
173 InvalidateWindowData(WC_STATION_VIEW, st->index, this->Cargo());
174 }
175}

199void Path::Fork(Path *base, uint cap, int free_cap, uint dist)
200{
201 this->capacity = std::min(base->capacity, cap);
202 this->free_capacity = std::min(base->free_capacity, free_cap);
203 this->distance = base->distance + dist;
204 assert(this->distance > 0);
205 if (this->parent != base) {
206 this->Detach();
207 this->parent = base;
208 this->parent->num_children++;
209 }
210 this->origin = base->origin;
211}

221uint Path::AddFlow(uint new_flow, LinkGraphJob &job, uint max_saturation)
222{
223 if (this->parent != nullptr) {
224 LinkGraphJob::EdgeAnnotation &edge = job[this->parent->node][this->node];
225 if (max_saturation != UINT_MAX) {
226 uint usable_cap = edge.base.capacity * max_saturation / 100;
227 if (usable_cap > edge.Flow()) {
228 new_flow = std::min(new_flow, usable_cap - edge.Flow());
229 } else {
230 return 0;
231 }
232 }
233 new_flow = this->parent->AddFlow(new_flow, job, max_saturation);
234 if (this->flow == 0 && new_flow > 0) {
235 job[this->parent->node].paths.push_front(this);
236 }
237 edge.AddFlow(new_flow);
238 }
239 this->flow += new_flow;
240 return new_flow;
241}
