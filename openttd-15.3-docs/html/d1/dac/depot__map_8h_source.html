53inline DepotID GetDepotIndex(Tile t)
54{
55 /* Hangars don't have a Depot class, thus store no DepotID. */
56 assert(IsRailDepotTile(t) || IsRoadDepotTile(t) || IsShipDepotTile(t));
57 return DepotID{t.m2()};
58}
