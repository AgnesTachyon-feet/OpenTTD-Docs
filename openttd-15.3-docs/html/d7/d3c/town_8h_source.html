116 inline SuppliedCargo &GetOrCreateCargoSupplied(CargoType cargo)
117 {
118 assert(IsValidCargoType(cargo));
119 auto it = std::ranges::lower_bound(this->supplied, cargo, std::less{}, &SuppliedCargo::cargo);
120 if (it == std::end(this->supplied) || it->cargo != cargo) it = this->supplied.emplace(it, cargo);
121 return *it;
122 }

124 inline SuppliedCargoes::const_iterator GetCargoSupplied(CargoType cargo) const
125 {
126 if (!IsValidCargoType(cargo)) return std::end(this->supplied);
127 auto it = std::ranges::lower_bound(this->supplied, cargo, std::less{}, &SuppliedCargo::cargo);
128 if (it == std::end(this->supplied) || it->cargo != cargo) return std::end(supplied);
129 return it;
130 }

132 inline uint8_t GetPercentTransported(CargoType cargo_type) const
133 {

135 if (it == std::end(this->supplied)) return 0;

283template <class T>
284void MakeDefaultName(T *obj)
285{
286 /* We only want to set names if it hasn't been set before, or when we're calling from afterload. */
287 assert(obj->name.empty() || obj->town_cn == UINT16_MAX);
288
289 obj->town = ClosestTownFromTile(obj->xy, UINT_MAX);
290
291 /* Find first unused number belonging to this town. This can never fail,
292 * as long as there can be at most 65535 waypoints/depots in total.
293 *
294 * This does 'n * m' search, but with 32bit 'used' bitmap, it needs at
295 * most 'n * (1 + ceil(m / 32))' steps (n - number of waypoints in pool,
296 * m - number of waypoints near this town).
297 * Usually, it needs only 'n' steps.
298 *
299 * If it wasn't using 'used' and 'idx', it would just search for increasing 'next',
300 * but this way it is faster */
301
302 uint32_t used = 0; // bitmap of used waypoint numbers, sliding window with 'next' as base
303 uint32_t next = 0; // first number in the bitmap
304 uint32_t idx = 0; // index where we will stop
305 uint32_t cid = 0; // current index, goes to T::GetPoolSize()-1, then wraps to 0
306
307 do {
308 T *lobj = T::GetIfValid(cid);
309
310 /* check only valid waypoints... */
311 if (lobj != nullptr && obj != lobj) {
312 /* only objects within the same city and with the same type */
313 if (lobj->town == obj->town && lobj->IsOfType(obj)) {
314 /* if lobj->town_cn < next, uint will overflow to '+inf' */
315 uint i = (uint)lobj->town_cn - next;
316
317 if (i < 32) {
318 SetBit(used, i); // update bitmap
319 if (i == 0) {
320 /* shift bitmap while the lowest bit is '1';
321 * increase the base of the bitmap too */
322 do {
323 used >>= 1;
324 next++;
325 } while (HasBit(used, 0));
326 /* when we are at 'idx' again at end of the loop and
327 * 'next' hasn't changed, then no object had town_cn == next,
328 * so we can safely use it */
329 idx = cid;
330 }
331 }
332 }
333 }
334
335 cid++;
336 if (cid == T::GetPoolSize()) cid = 0; // wrap to zero...
337 } while (cid != idx);
338
339 obj->town_cn = (uint16_t)next; // set index...
340}

342/*
343 * Converts original town ticks counters to plain game ticks. Note that
344 * tick 0 is a valid tick so actual amount is one more than the counter value.
345 */
346inline uint16_t TownTicksToGameTicks(uint16_t ticks)
347{
348 return (std::min(ticks, MAX_TOWN_GROWTH_TICKS) + 1) * Ticks::TOWN_GROWTH_TICKS - 1;
349}
