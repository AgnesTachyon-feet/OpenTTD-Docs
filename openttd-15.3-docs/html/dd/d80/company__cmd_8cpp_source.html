284static void SubtractMoneyFromAnyCompany(Company *c, const CommandCost &cost)
285{
286 if (cost.GetCost() == 0) return;
287 assert(cost.GetExpensesType() != INVALID_EXPENSES);
288
289 c->money -= cost.GetCost();
290 c->yearly_expenses[0][cost.GetExpensesType()] += cost.GetCost();
291
292 if (HasBit(1 << EXPENSES_TRAIN_REVENUE |
293 1 << EXPENSES_ROADVEH_REVENUE |
294 1 << EXPENSES_AIRCRAFT_REVENUE |
295 1 << EXPENSES_SHIP_REVENUE, cost.GetExpensesType())) {
296 c->cur_economy.income -= cost.GetCost();
297 } else if (HasBit(1 << EXPENSES_TRAIN_RUN |
298 1 << EXPENSES_ROADVEH_RUN |
299 1 << EXPENSES_AIRCRAFT_RUN |
300 1 << EXPENSES_SHIP_RUN |
301 1 << EXPENSES_PROPERTY |
302 1 << EXPENSES_LOAN_INTEREST, cost.GetExpensesType())) {
303 c->cur_economy.expenses -= cost.GetCost();
304 }
305
306 InvalidateCompanyWindows(c);
307}

324void SubtractMoneyFromCompanyFract(CompanyID company, const CommandCost &cst)
325{
326 Company *c = Company::Get(company);
327 uint8_t m = c->money_fraction;
328 Money cost = cst.GetCost();
329
330 c->money_fraction = m - (uint8_t)cost;
331 cost >>= 8;
332 if (c->money_fraction > m) cost++;
333 if (cost != 0) SubtractMoneyFromAnyCompany(c, CommandCost(cst.GetExpensesType(), cost));
334}

413static void GenerateCompanyName(Company *c)
414{
415 if (c->name_1 != STR_SV_UNNAMED) return;
416 if (c->last_build_coordinate == 0) return;
417
418 Town *t = ClosestTownFromTile(c->last_build_coordinate, UINT_MAX);
419
420 StringID str;
421 uint32_t strp;
422 std::string name;
423 if (t->name.empty() && IsInsideMM(t->townnametype, SPECSTR_TOWNNAME_START, SPECSTR_TOWNNAME_END)) {
424 str = t->townnametype - SPECSTR_TOWNNAME_START + SPECSTR_COMPANY_NAME_START;
425 strp = t->townnameparts;
426
427verify_name:;
428 /* No companies must have this name already */
429 for (const Company *cc : Company::Iterate()) {
430 if (cc->name_1 == str && cc->name_2 == strp) goto bad_town_name;
431 }
432
433 name = GetString(str, strp);
434 if (Utf8StringLength(name) >= MAX_LENGTH_COMPANY_NAME_CHARS) goto bad_town_name;
435
436set_name:;
437 c->name_1 = str;
438 c->name_2 = strp;
439
440 MarkWholeScreenDirty();
441 AI::BroadcastNewEvent(new ScriptEventCompanyRenamed(c->index, name));
442 Game::NewEvent(new ScriptEventCompanyRenamed(c->index, name));
443
444 if (c->is_ai) {
445 auto cni = std::make_unique<CompanyNewsInformation>(STR_NEWS_COMPANY_LAUNCH_TITLE, c);
446 EncodedString headline = GetEncodedString(STR_NEWS_COMPANY_LAUNCH_DESCRIPTION, cni->company_name, t->index);
447 AddNewsItem(std::move(headline),
448 NewsType::CompanyInfo, NewsStyle::Company, {}, c->last_build_coordinate, {}, std::move(cni));
449 }
450 return;
451 }
452bad_town_name:;
453
454 if (c->president_name_1 == SPECSTR_PRESIDENT_NAME) {
455 str = SPECSTR_ANDCO_NAME;
456 strp = c->president_name_2;
457 name = GetString(str, strp);
458 goto set_name;
459 } else {
460 str = SPECSTR_ANDCO_NAME;
461 strp = Random();
462 goto verify_name;
463 }
464}

467static const uint8_t _colour_sort[COLOUR_END] = {2, 2, 3, 2, 3, 2, 3, 2, 3, 2, 2, 2, 3, 1, 1, 1};
469static const Colours _similar_colour[COLOUR_END][2] = {
470 { COLOUR_BLUE, COLOUR_LIGHT_BLUE }, // COLOUR_DARK_BLUE
471 { COLOUR_GREEN, COLOUR_DARK_GREEN }, // COLOUR_PALE_GREEN

477 { COLOUR_PALE_GREEN, COLOUR_GREEN }, // COLOUR_DARK_GREEN

482 { COLOUR_YELLOW, COLOUR_CREAM }, // COLOUR_ORANGE
483 { COLOUR_CREAM, INVALID_COLOUR }, // COLOUR_BROWN

492static Colours GenerateCompanyColour()
493{
494 Colours colours[COLOUR_END];
495
496 /* Initialize array */
497 for (uint i = 0; i < COLOUR_END; i++) colours[i] = static_cast<Colours>(i);
498
499 /* And randomize it */
500 for (uint i = 0; i < 100; i++) {
501 uint r = Random();
502 std::swap(colours[GB(r, 0, 4)], colours[GB(r, 4, 4)]);
503 }
504
505 /* Bubble sort it according to the values in table 1 */
506 for (uint i = 0; i < COLOUR_END; i++) {
507 for (uint j = 1; j < COLOUR_END; j++) {
508 if (_colour_sort[colours[j - 1]] < _colour_sort[colours[j]]) {
509 std::swap(colours[j - 1], colours[j]);
510 }
511 }
512 }
513
514 /* Move the colours that look similar to each company's colour to the side */
515 for (const Company *c : Company::Iterate()) {
516 Colours pcolour = c->colour;
517
518 for (uint i = 0; i < COLOUR_END; i++) {
519 if (colours[i] == pcolour) {
520 colours[i] = INVALID_COLOUR;
521 break;
522 }
523 }
524
525 for (uint j = 0; j < 2; j++) {
526 Colours similar = _similar_colour[pcolour][j];
527 if (similar == INVALID_COLOUR) break;
528
529 for (uint i = 1; i < COLOUR_END; i++) {
530 if (colours[i - 1] == similar) std::swap(colours[i - 1], colours[i]);
531 }
532 }
533 }
534
535 /* Return the first available colour */
536 for (uint i = 0; i < COLOUR_END; i++) {
537 if (colours[i] != INVALID_COLOUR) return colours[i];
538 }
539
540 NOT_REACHED();
541}

547static void GeneratePresidentName(Company *c)
548{
549 for (;;) {
550restart:;
551 c->president_name_2 = Random();
552 c->president_name_1 = SPECSTR_PRESIDENT_NAME;
553
554 /* Reserve space for extra unicode character. We need to do this to be able
555 * to detect too long president name. */
556 std::string name = GetString(STR_PRESIDENT_NAME, c->index);
557 if (Utf8StringLength(name) >= MAX_LENGTH_PRESIDENT_NAME_CHARS) continue;

560 if (c != cc) {
561 std::string other_name = GetString(STR_PRESIDENT_NAME, cc->index);
562 if (name == other_name) goto restart;
563 }
564 }
565 return;
566 }

598Company *DoStartupNewCompany(bool is_ai, CompanyID company = CompanyID::Invalid())
599{
600 if (!Company::CanAllocateItem()) return nullptr;
601
602 /* we have to generate colour before this company is valid */
603 Colours colour = GenerateCompanyColour();
604
605 Company *c;
606 if (company == CompanyID::Invalid()) {
607 c = new Company(STR_SV_UNNAMED, is_ai);
608 } else {
609 if (Company::IsValidID(company)) return nullptr;
610 c = new (company) Company(STR_SV_UNNAMED, is_ai);
611 }
612
613 c->colour = colour;
614
615 ResetCompanyLivery(c);
616 _company_colours[c->index] = c->colour;
617
618 /* Scale the initial loan based on the inflation rounded down to the loan interval. The maximum loan has already been inflation adjusted. */
619 c->money = c->current_loan = std::min<int64_t>((INITIAL_LOAN * _economy.inflation_prices >> 16) / LOAN_INTERVAL * LOAN_INTERVAL, _economy.max_loan);
620
621 c->avail_railtypes = GetCompanyRailTypes(c->index);
622 c->avail_roadtypes = GetCompanyRoadTypes(c->index);
623 c->inaugurated_year = TimerGameEconomy::year;
624 c->inaugurated_year_calendar = TimerGameCalendar::year;
625
626 /* If starting a player company in singleplayer and a favourite company manager face is selected, choose it. Otherwise, use a random face.
627 * In a network game, we'll choose the favourite face later in CmdCompanyCtrl to sync it to all clients. */
628 bool randomise_face = true;
629 if (!_company_manager_face.empty() && !is_ai && !_networking) {
630 auto cmf = ParseCompanyManagerFaceCode(_company_manager_face);
631 if (cmf.has_value()) {
632 randomise_face = false;
633 c->face = std::move(*cmf);
634 }
635 }
636 if (randomise_face) RandomiseCompanyManagerFace(c->face, _random);
637
638 SetDefaultCompanySettings(c->index);
639 ClearEnginesHiddenFlagOfCompany(c->index);
640
641 GeneratePresidentName(c);
642
643 SetWindowDirty(WC_GRAPH_LEGEND, 0);
644 InvalidateWindowData(WC_CLIENT_LIST, 0);
645 InvalidateWindowData(WC_LINKGRAPH_LEGEND, 0);
646 BuildOwnerLegend();
647 InvalidateWindowData(WC_SMALLMAP, 0, 1);
648
649 if (is_ai && (!_networking || _network_server)) AI::StartNew(c->index);
650
651 AI::BroadcastNewEvent(new ScriptEventCompanyNew(c->index), c->index);
652 Game::NewEvent(new ScriptEventCompanyNew(c->index));
653
654 return c;
655}

716static void HandleBankruptcyTakeover(Company *c)
717{
718 /* Amount of time out for each company to take over a company;
719 * Timeout is a quarter (3 months of 30 days) divided over the
720 * number of companies. The minimum number of days in a quarter
721 * is 90: 31 in January, 28 in February and 31 in March.
722 * Note that the company going bankrupt can't buy itself. */
723 static const int TAKE_OVER_TIMEOUT = 3 * 30 * Ticks::DAY_TICKS / (MAX_COMPANIES - 1);
724
725 assert(c->bankrupt_asked.Any());
726
727 /* We're currently asking some company to buy 'us' */
728 if (c->bankrupt_timeout != 0) {
729 c->bankrupt_timeout -= MAX_COMPANIES;
730 if (c->bankrupt_timeout > 0) return;
731 c->bankrupt_timeout = 0;
732
733 return;
734 }
735
736 /* Did we ask everyone for bankruptcy? If so, bail out. */
737 if (c->bankrupt_asked.All()) return;
738
739 Company *best = nullptr;
740 int32_t best_performance = -1;
741
742 /* Ask the company with the highest performance history first */
743 for (Company *c2 : Company::Iterate()) {
744 if (c2->bankrupt_asked.None() && // Don't ask companies going bankrupt themselves
745 !c->bankrupt_asked.Test(c2->index) &&
746 best_performance < c2->old_economy[1].performance_history &&
747 CheckTakeoverVehicleLimit(c2->index, c->index)) {
748 best_performance = c2->old_economy[1].performance_history;
749 best = c2;
750 }
751 }
752
753 /* Asked all companies? */
754 if (best_performance == -1) {
755 c->bankrupt_asked.Set();
756 return;
757 }
758
759 c->bankrupt_asked.Set(best->index);
760
761 c->bankrupt_timeout = TAKE_OVER_TIMEOUT;
762
763 AI::NewEvent(best->index, new ScriptEventCompanyAskMerger(c->index, c->bankrupt_value));
764 if (IsInteractiveCompany(best->index)) {
765 ShowBuyCompanyDialog(c->index, false);
766 }
767}

1491{
1492 if (str.empty()) return std::nullopt;
1493
1494 CompanyManagerFace cmf;
1495 StringConsumer consumer{str};
1496 if (consumer.FindChar(':') != StringConsumer::npos) {
1497 auto label = consumer.ReadUntilChar(':', StringConsumer::SKIP_ONE_SEPARATOR);
1498
1499 /* Read numeric part and ensure it's valid. */
1500 auto bits = consumer.TryReadIntegerBase<uint32_t>(10, true);
1501 if (!bits.has_value() || consumer.AnyBytesLeft()) return std::nullopt;
1502
1503 /* Ensure style label is valid. */
1504 auto style = FindCompanyManagerFaceLabel(label);
1505 if (!style.has_value()) return std::nullopt;
1506
1507 SetCompanyManagerFaceStyle(cmf, *style);
1508 cmf.bits = *bits;
1509 } else {
1510 /* No ':' included, treat as numeric-only. This allows old-style codes to be entered. */
1511 auto bits = ParseInteger(str, 10, true);
1512 if (!bits.has_value()) return std::nullopt;
1513
1514 /* Old codes use bits 0..1 to represent face style. These map directly to the default face styles. */
1515 SetCompanyManagerFaceStyle(cmf, GB(*bits, 0, 2));
1516 cmf.bits = *bits;
1517 }
1518
1519 /* Force the face bits to be valid. */
1520 FaceVars vars = GetCompanyManagerFaceVars(cmf.style);
1521 ScaleAllCompanyManagerFaceBits(cmf, vars);
1522 cmf.bits = MaskCompanyManagerFaceBits(cmf, vars);
1523
1524 return cmf;
