797inline constexpr size_t SlVarSize(VarType type)
798{
799 switch (GetVarMemType(type)) {
800 case SLE_VAR_BL: return sizeof(bool);
801 case SLE_VAR_I8: return sizeof(int8_t);
802 case SLE_VAR_U8: return sizeof(uint8_t);
803 case SLE_VAR_I16: return sizeof(int16_t);
804 case SLE_VAR_U16: return sizeof(uint16_t);
805 case SLE_VAR_I32: return sizeof(int32_t);
806 case SLE_VAR_U32: return sizeof(uint32_t);
807 case SLE_VAR_I64: return sizeof(int64_t);
808 case SLE_VAR_U64: return sizeof(uint64_t);
809 case SLE_VAR_NULL: return sizeof(void *);
810 case SLE_VAR_STR: return sizeof(std::string);
811 case SLE_VAR_STRQ: return sizeof(std::string);
812 case SLE_VAR_NAME: return sizeof(std::string);
813 default: NOT_REACHED();
814 }
815}

825inline constexpr bool SlCheckVarSize(SaveLoadType cmd, VarType type, size_t length, size_t size)
826{
827 switch (cmd) {
828 case SL_VAR: return SlVarSize(type) == size;
829 case SL_REF: return sizeof(void *) == size;
830 case SL_STDSTR: return SlVarSize(type) == size;
831 case SL_ARR: return SlVarSize(type) * length <= size; // Partial load of array is permitted.
832 case SL_DEQUE: return sizeof(std::deque<void *>) == size;
833 case SL_VECTOR: return sizeof(std::vector<void *>) == size;
834 case SL_REFLIST: return sizeof(std::list<void *>) == size;
835 case SL_REFVECTOR: return sizeof(std::vector<void *>) == size;
836 case SL_SAVEBYTE: return true;
837 default: NOT_REACHED();
838 }
839}
