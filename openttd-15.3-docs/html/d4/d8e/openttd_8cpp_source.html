119void UserErrorI(const std::string &str)
120{
121 ShowOSErrorBox(str, false);
122 if (VideoDriver::GetInstance() != nullptr) VideoDriver::GetInstance()->Stop();
123
124#ifdef __EMSCRIPTEN__
125 emscripten_exit_pointerlock();
126 /* In effect, the game ends here. As emscripten_set_main_loop() caused
127 * the stack to be unwound, the code after MainLoop() in
128 * openttd_main() is never executed. */
129 EM_ASM(if (window["openttd_abort"]) openttd_abort());
130#endif
131
132 _exit(1);
133}

230static void WriteSavegameInfo(const std::string &name)
231{
232 extern SaveLoadVersion _sl_version;
233 uint32_t last_ottd_rev = 0;
234 uint8_t ever_modified = 0;
235 bool removed_newgrfs = false;
236
237 _gamelog.Info(&last_ottd_rev, &ever_modified, &removed_newgrfs);
238
239 std::string message;
240 message.reserve(1024);
241 format_append(message, "Name: {}\n", name);
242 format_append(message, "Savegame ver: {}\n", _sl_version);
243 format_append(message, "NewGRF ver: 0x{:08X}\n", last_ottd_rev);
244 format_append(message, "Modified: {}\n", ever_modified);
245
246 if (removed_newgrfs) {
247 format_append(message, "NewGRFs have been removed\n");
248 }
249
250 message += "NewGRFs:\n";
251 if (_load_check_data.HasNewGrfs()) {
252 for (const auto &c : _load_check_data.grfconfig) {
253 format_append(message, "{:08X} {} {}\n", std::byteswap(c->ident.grfid),
254 FormatArrayAsHex(c->flags.Test(GRFConfigFlag::Compatible) ? c->original_md5sum : c->ident.md5sum), c->filename);
255 }
256 }
257
258 /* ShowInfo put output to stderr, but version information should go
259 * to stdout; this is the only exception */
260#if !defined(_WIN32)
261 fmt::print("{}\n", message);
262#else
263 ShowInfoI(message);
264#endif
265}

274static void ParseResolution(Dimension &res, std::string_view s)
275{
276 StringConsumer consumer(s);
277 auto width = consumer.TryReadIntegerBase<uint>(10);
278 auto valid = consumer.ReadIf("x");
279 auto height = consumer.TryReadIntegerBase<uint>(10);
280 if (!width.has_value() || !valid || !height.has_value() || consumer.AnyBytesLeft()) {
281 ShowInfo("Invalid resolution '{}'", s);
282 return;
283 }
284
285 res.width = std::max<uint>(*width, 64);
286 res.height = std::max<uint>(*height, 64);
287}

374struct AfterNewGRFScan : NewGRFScanCallback {
375 TimerGameCalendar::Year startyear = CalendarTime::INVALID_YEAR;
376 uint32_t generation_seed = GENERATE_NEW_SEED;
377 std::string dedicated_host;
378 uint16_t dedicated_port = 0;
379 std::string connection_string;
380 std::string join_server_password;
381 bool save_config = true;
382
386 AfterNewGRFScan()
387 {
388 /* Visual C++ 2015 fails compiling this line (AfterNewGRFScan::generation_seed undefined symbol)
389 * if it's placed outside a member function, directly in the struct body. */
390 static_assert(sizeof(generation_seed) == sizeof(_settings_game.game_creation.generation_seed));
391 }
392
393 void OnNewGRFsScanned() override
394 {
395 ResetGRFConfig(false);
396
397 TarScanner::DoScan(TarScanner::Mode::Scenario);
398
399 AI::Initialize();
400 Game::Initialize();
401
402 /* We want the new (correct) NewGRF count to survive the loading. */
403 uint last_newgrf_count = _settings_client.gui.last_newgrf_count;
404 LoadFromConfig();
405 _settings_client.gui.last_newgrf_count = last_newgrf_count;
406 /* Since the default for the palette might have changed due to
407 * reading the configuration file, recalculate that now. */
408 UpdateNewGRFConfigPalette();
409
410 Game::Uninitialize(true);
411 AI::Uninitialize(true);
412 LoadFromHighScore();
413 LoadHotkeysFromConfig();
414 WindowDesc::LoadFromConfig();
415
416 /* We have loaded the config, so we may possibly save it. */
417 _save_config = save_config;
418
419 /* restore saved music and effects volumes */
420 MusicDriver::GetInstance()->SetVolume(_settings_client.music.music_vol);
421 SetEffectVolume(_settings_client.music.effect_vol);
422
423 if (startyear != CalendarTime::INVALID_YEAR) IConsoleSetSetting("game_creation.starting_year", startyear.base());
424 _settings_newgame.game_creation.generation_seed = generation_seed;
425
426 if (!dedicated_host.empty()) {
427 _network_bind_list.clear();
428 _network_bind_list.emplace_back(dedicated_host);
429 }
430 if (dedicated_port != 0) _settings_client.network.server_port = dedicated_port;
431
432 /* initialize the ingame console */
433 IConsoleInit();
434 InitializeGUI();
435 IConsoleCmdExec("exec scripts/autoexec.scr 0");
436
437 /* Make sure _settings is filled with _settings_newgame if we switch to a game directly */
438 if (_switch_mode != SM_NONE) MakeNewgameSettingsLive();
439
440 if (_network_available && !connection_string.empty()) {
441 LoadIntroGame();
442 _switch_mode = SM_NONE;
443
444 NetworkClientConnectGame(connection_string, COMPANY_NEW_COMPANY, join_server_password);
445 }
446
447 /* After the scan we're not used anymore. */
448 delete this;
449 }
450};

500int openttd_main(std::span<std::string_view> arguments)
501{
502 _game_session_stats.start_time = std::chrono::steady_clock::now();
503 _game_session_stats.savegame_size = std::nullopt;
504
505 std::string musicdriver;
506 std::string sounddriver;
507 std::string videodriver;
508 std::string blitter;
509 std::string graphics_set;
510 std::string sounds_set;
511 std::string music_set;
512 Dimension resolution = {0, 0};
513 std::unique_ptr<AfterNewGRFScan> scanner = std::make_unique<AfterNewGRFScan>();
514 bool dedicated = false;
515 bool only_local_path = false;
516
517 extern bool _dedicated_forks;
518 _dedicated_forks = false;
519
520 _game_mode = GM_MENU;
521 _switch_mode = SM_MENU;
522
523 auto options = CreateOptions();
524 GetOptData mgo(arguments.subspan(1), options);
525
526 int i;
527 while ((i = mgo.GetOpt()) != -1) {
528 switch (i) {
529 case 'I': graphics_set = mgo.opt; break;
530 case 'S': sounds_set = mgo.opt; break;
531 case 'M': music_set = mgo.opt; break;
532 case 'm': musicdriver = mgo.opt; break;
533 case 's': sounddriver = mgo.opt; break;
534 case 'v': videodriver = mgo.opt; break;
535 case 'b': blitter = mgo.opt; break;
536 case 'D':
537 musicdriver = "null";
538 sounddriver = "null";
539 videodriver = "dedicated";
540 blitter = "null";
541 dedicated = true;
542 SetDebugString("net=4", ShowInfoI);
543 if (!mgo.opt.empty()) {
544 scanner->dedicated_host = ParseFullConnectionString(mgo.opt, scanner->dedicated_port);
545 }
546 break;
547 case 'f': _dedicated_forks = true; break;
548 case 'n':
549 scanner->connection_string = mgo.opt; // host:port#company parameter
550 break;
551 case 'p':
552 scanner->join_server_password = mgo.opt;
553 break;
554 case 'r': ParseResolution(resolution, mgo.opt); break;
555 case 't':
556 if (auto value = ParseInteger(mgo.opt); value.has_value()) {
557 scanner->startyear = TimerGameCalendar::Year(*value);
558 } else {
559 fmt::print(stderr, "Invalid start year: {}\n", mgo.opt);
560 }
561 break;
562 case 'd': {
563#if defined(_WIN32)
564 CreateConsole();
565#endif
566 if (!mgo.opt.empty()) SetDebugString(mgo.opt, ShowInfoI);
567 break;
568 }
569 case 'e':
570 /* Allow for '-e' before or after '-g'. */
571 switch (_switch_mode) {
572 case SM_MENU: _switch_mode = SM_EDITOR; break;
573 case SM_LOAD_GAME: _switch_mode = SM_LOAD_SCENARIO; break;
574 case SM_START_HEIGHTMAP: _switch_mode = SM_LOAD_HEIGHTMAP; break;
575 default: break;
576 }
577 break;
578 case 'g':
579 if (!mgo.opt.empty()) {
580 _file_to_saveload.name = mgo.opt;
581
582 std::string extension = FS2OTTD(std::filesystem::path(OTTD2FS(_file_to_saveload.name)).extension().native());
583 auto [ft, _] = FiosGetSavegameListCallback(SLO_LOAD, _file_to_saveload.name, extension);
584 if (ft == FIOS_TYPE_INVALID) {
585 std::tie(ft, _) = FiosGetScenarioListCallback(SLO_LOAD, _file_to_saveload.name, extension);
586 }
587 if (ft == FIOS_TYPE_INVALID) {
588 std::tie(ft, _) = FiosGetHeightmapListCallback(SLO_LOAD, _file_to_saveload.name, extension);
589 }
590
591 /* Allow for '-e' before or after '-g'. */
592 switch (ft.abstract) {
593 case FT_SAVEGAME: _switch_mode = (_switch_mode == SM_EDITOR ? SM_LOAD_SCENARIO : SM_LOAD_GAME); break;
594 case FT_SCENARIO: _switch_mode = (_switch_mode == SM_EDITOR ? SM_LOAD_SCENARIO : SM_LOAD_GAME); break;
595 case FT_HEIGHTMAP: _switch_mode = (_switch_mode == SM_EDITOR ? SM_LOAD_HEIGHTMAP : SM_START_HEIGHTMAP); break;
596 default: break;
597 }
598
599 _file_to_saveload.SetMode(ft, SLO_LOAD);
600 break;
601 }
602
603 _switch_mode = SM_NEWGAME;
604 /* Give a random map if no seed has been given */
605 if (scanner->generation_seed == GENERATE_NEW_SEED) {
606 scanner->generation_seed = InteractiveRandom();
607 }
608 break;
609 case 'q': {
610 DeterminePaths(arguments[0], only_local_path);
611 if (mgo.opt.empty()) {
612 return 1;
613 }
614
615 std::string extension = FS2OTTD(std::filesystem::path(OTTD2FS(mgo.opt)).extension().native());
616 auto [_, title] = FiosGetSavegameListCallback(SLO_LOAD, mgo.opt, extension);
617
618 _load_check_data.Clear();
619 SaveOrLoadResult res = SaveOrLoad(mgo.opt, SLO_CHECK, DFT_GAME_FILE, SAVE_DIR, false);
620 if (res != SL_OK || _load_check_data.HasErrors()) {
621 fmt::print(stderr, "Failed to open savegame\n");
622 if (_load_check_data.HasErrors()) {
623 InitializeLanguagePacks(); // A language pack is needed for GetString()
624 fmt::print(stderr, "{}\n", GetString(_load_check_data.error, _load_check_data.error_msg));
625 }
626 return 1;
627 }
628
629 WriteSavegameInfo(title);
630 return 0;
631 }
632 case 'Q': {
633 extern int _skip_all_newgrf_scanning;
634 _skip_all_newgrf_scanning += 1;
635 break;
636 }
637 case 'G':
638 if (auto value = ParseInteger(mgo.opt); value.has_value()) {
639 scanner->generation_seed = *value;
640 } else {
641 fmt::print(stderr, "Invalid generation seed: {}\n", mgo.opt);
642 }
643 break;
644 case 'c': _config_file = mgo.opt; break;
645 case 'x': scanner->save_config = false; break;
646 case 'X': only_local_path = true; break;
647 case 'h': break; // handled below
648 }
649 if (i == 'h' || i == -2) break;
650 }
651
652 if (i == 'h' || i == -2 || !mgo.arguments.empty()) {
653 /* Either the user typed '-h', they made an error, or they added unrecognized command line arguments.
654 * In all cases, print the help, and exit.
655 *
656 * The next two functions are needed to list the graphics sets. We can't do them earlier
657 * because then we cannot show it on the debug console as that hasn't been configured yet. */
658 DeterminePaths(arguments[0], only_local_path);
659 TarScanner::DoScan(TarScanner::Mode::Baseset);
660 BaseGraphics::FindSets();
661 BaseSounds::FindSets();
662 BaseMusic::FindSets();
663 ShowHelp();
664
665 /* Return zero when asked to print help, and 1 for all other cases. */
666 if (i == 'h') return 0;
667 return 1;
668 }
669
670 DeterminePaths(arguments[0], only_local_path);
671 TarScanner::DoScan(TarScanner::Mode::Baseset);
672
673 if (dedicated) Debug(net, 3, "Starting dedicated server, version {}", _openttd_revision);
674 if (_dedicated_forks && !dedicated) _dedicated_forks = false;
675
676#if defined(UNIX)
677 /* We must fork here, or we'll end up without some resources we need (like sockets) */
678 if (_dedicated_forks) DedicatedFork();
679#endif
680
681 LoadFromConfig(true);
682
683 if (resolution.width != 0) _cur_resolution = resolution;
684
685 /* Limit width times height times bytes per pixel to fit a 32 bit
686 * integer, This way all internal drawing routines work correctly.
687 * A resolution that has one component as 0 is treated as a marker to
688 * auto-detect a good window size. */
689 _cur_resolution.width = std::min(_cur_resolution.width, UINT16_MAX / 2u);
690 _cur_resolution.height = std::min(_cur_resolution.height, UINT16_MAX / 2u);
691
692 /* Assume the cursor starts within the game as not all video drivers
693 * get an event that the cursor is within the window when it is opened.
694 * Saying the cursor is there makes no visible difference as it would
695 * just be out of the bounds of the window. */
696 _cursor.in_window = true;
697
698 /* enumerate language files */
699 InitializeLanguagePacks();
700
701 /* Initialize the font cache */
702 FontCache::LoadFontCaches(FONTSIZES_REQUIRED);
703
704 /* This must be done early, since functions use the SetWindowDirty* calls */
705 InitWindowSystem();
706
707 BaseGraphics::FindSets();
708 bool valid_graphics_set;
709 if (!graphics_set.empty()) {
710 valid_graphics_set = BaseGraphics::SetSetByName(graphics_set);
711 } else if (BaseGraphics::ini_data.shortname != 0) {
712 graphics_set = BaseGraphics::ini_data.name;
713 valid_graphics_set = BaseGraphics::SetSetByShortname(BaseGraphics::ini_data.shortname);
714 if (valid_graphics_set && !BaseGraphics::ini_data.extra_params.empty()) {
715 GRFConfig &extra_cfg = BaseGraphics::GetUsedSet()->GetOrCreateExtraConfig();

941bool SafeLoad(const std::string &filename, SaveLoadOperation fop, DetailedFileType dft, GameMode newgm, Subdirectory subdir, std::shared_ptr<LoadFilter> lf = nullptr)
942{
943 assert(fop == SLO_LOAD);
944 assert(dft == DFT_GAME_FILE || (lf == nullptr && dft == DFT_OLD_GAME_FILE));
945 GameMode ogm = _game_mode;
946
947 _game_mode = newgm;
948
949 SaveOrLoadResult result = (lf == nullptr) ? SaveOrLoad(filename, fop, dft, subdir) : LoadWithFilter(std::move(lf));
950 if (result == SL_OK) return true;
951
952 if (_network_dedicated && ogm == GM_MENU) {
953 /*
954 * If we are a dedicated server *and* we just were in the menu, then we
955 * are loading the first savegame. If that fails, not starting the
956 * server is a better reaction than starting the server with a newly
957 * generated map as it is quite likely to be started from a script.
958 */
959 Debug(net, 0, "Loading requested map failed; closing server.");
960 _exit_game = true;
961 return false;
962 }
963
964 if (result != SL_REINIT) {
965 _game_mode = ogm;
966 return false;
967 }
968
969 if (_network_dedicated) {
970 /*
971 * If we are a dedicated server, have already loaded/started a game,
972 * and then loading the savegame fails in a manner that we need to
973 * reinitialize everything. We must not fall back into the menu mode
974 * with the intro game, as that is unjoinable by clients. So there is
975 * nothing else to do than start a new game, as it might have failed
976 * trying to reload the originally loaded savegame/scenario.
977 */
978 Debug(net, 0, "Loading game failed, so a new (random) game will be started");
979 MakeNewGame(false, true);
980 return false;
981 }
982
983 if (_network_server) {
984 /* We can't load the intro game as server, so disconnect first. */
985 NetworkDisconnect();
986 }
987
988 switch (ogm) {
989 default:
990 case GM_MENU: LoadIntroGame(); break;
991 case GM_EDITOR: MakeNewEditorWorld(); break;
992 }
993 return false;
994}

1018void SwitchToMode(SwitchMode new_mode)
1019{
1020 /* If we are saving something, the network stays in its current state */
1021 if (new_mode != SM_SAVE_GAME) {
1022 /* If the network is active, make it not-active */
1023 if (_networking) {
1024 if (_network_server && (new_mode == SM_LOAD_GAME || new_mode == SM_NEWGAME || new_mode == SM_RESTARTGAME)) {
1025 NetworkReboot();
1026 } else {
1027 NetworkDisconnect();
1028 }
1029 }
1030
1031 /* If we are a server, we restart the server */
1032 if (_is_network_server) {
1033 /* But not if we are going to the menu */
1034 if (new_mode != SM_MENU) {
1035 /* check if we should reload the config */
1036 if (_settings_client.network.reload_cfg) {
1037 LoadFromConfig();
1038 MakeNewgameSettingsLive();
1039 ResetGRFConfig(false);
1040 }
1041 NetworkServerStart();
1042 } else {
1043 /* This client no longer wants to be a network-server */
1044 _is_network_server = false;
1045 }
1046 }
1047 }
1048
1049 /* Make sure all AI controllers are gone at quitting game */
1050 if (new_mode != SM_SAVE_GAME) AI::KillAll();
1051
1052 /* When we change mode, reset the autosave. */
1053 if (new_mode != SM_SAVE_GAME) ChangeAutosaveFrequency(true);
1054

1056 if (_game_mode == GM_NORMAL && new_mode != SM_SAVE_GAME) _survey.Transmit(NetworkSurveyHandler::Reason::LEAVE);
1057
1058 /* Keep track when we last switch mode. Used for survey, to know how long someone was in a game. */

1060 _game_session_stats.start_time = std::chrono::steady_clock::now();
1061 _game_session_stats.savegame_size = std::nullopt;
1062 }

1064 switch (new_mode) {
1065 case SM_EDITOR: // Switch to scenario editor
1066 MakeNewEditorWorld();
1067 GenerateSavegameId();
1068
1069 UpdateSocialIntegration(GM_EDITOR);
1070 break;
1071
1072 case SM_RELOADGAME: // Reload with what-ever started the game
1073 if (_file_to_saveload.ftype.abstract == FT_SAVEGAME || _file_to_saveload.ftype.abstract == FT_SCENARIO) {

1082 break;
1083 }

1086 GenerateSavegameId();
1087
1088 UpdateSocialIntegration(GM_NORMAL);
1089 break;
1090
1091 case SM_RESTARTGAME: // Restart --> 'Random game' with current settings
1092 case SM_NEWGAME: // New Game --> 'Random game'
1093 MakeNewGame(false, new_mode == SM_NEWGAME);
1094 GenerateSavegameId();
1095
1096 UpdateSocialIntegration(GM_NORMAL);
1097 break;
1098
1099 case SM_LOAD_GAME: { // Load game, Play Scenario
1100 ResetGRFConfig(true);
1101 ResetWindowSystem();
1102
1103 if (!SafeLoad(_file_to_saveload.name, _file_to_saveload.file_op, _file_to_saveload.ftype.detailed, GM_NORMAL, NO_DIRECTORY)) {

1114 UpdateSocialIntegration(GM_NORMAL);
1115 break;
1116 }
1117
1118 case SM_RESTART_HEIGHTMAP: // Load a heightmap and start a new game from it with current settings
1119 case SM_START_HEIGHTMAP: // Load a heightmap and start a new game from it
1120 MakeNewGame(true, new_mode == SM_START_HEIGHTMAP);
1121 GenerateSavegameId();
1122
1123 UpdateSocialIntegration(GM_NORMAL);
1124 break;
1125
1126 case SM_LOAD_HEIGHTMAP: // Load heightmap from scenario editor
1127 SetLocalCompany(OWNER_NONE);
1128
1129 _game_mode = GM_EDITOR;
1130

1146 ShowErrorMessage(GetSaveLoadErrorType(), GetSaveLoadErrorMessage(), WL_CRITICAL);
1147 }
1148
1149 UpdateSocialIntegration(GM_EDITOR);
1150 break;
1151 }
1152
1153 case SM_JOIN_GAME: // Join a multiplayer game
1154 LoadIntroGame();
1155 NetworkClientJoinGame();
1156
1157 SocialIntegration::EventJoiningMultiplayer();
1158 break;

1162 if (BaseSounds::ini_set.empty() && BaseSounds::GetUsedSet()->fallback && SoundDriver::GetInstance()->HasOutput()) {

1171 ShowNetworkAskSurvey();

1174
1175 UpdateSocialIntegration(GM_MENU);
1176 break;
1177
1178 case SM_SAVE_GAME: // Save game.
1179 /* Make network saved games on pause compatible to singleplayer mode */
1180 if (SaveOrLoad(_file_to_saveload.name, SLO_SAVE, DFT_GAME_FILE, NO_DIRECTORY) != SL_OK) {
1181 ShowErrorMessage(GetSaveLoadErrorType(), GetSaveLoadErrorMessage(), WL_ERROR);

1192 case SM_GENRANDLAND: // Generate random land within scenario editor
1193 SetLocalCompany(OWNER_NONE);
1194 GenerateWorld(GWM_RANDOM, 1 << _settings_game.game_creation.map_x, 1 << _settings_game.game_creation.map_y);

1336void GameLoop()
1337{
1338 if (_game_mode == GM_BOOTSTRAP) {
1339 /* Check for UDP stuff */
1340 if (_network_available) NetworkBackgroundLoop();
1341 return;
1342 }
1343
1344 if (_request_newgrf_scan) {
1345 ScanNewGRFFiles(_request_newgrf_scan_callback);
1346 _request_newgrf_scan = false;
1347 _request_newgrf_scan_callback = nullptr;
1348 /* In case someone closed the game during our scan, don't do anything else. */
1349 if (_exit_game) return;
1350 }
1351
1352 ProcessAsyncSaveFinish();
1353
1354 if (_game_mode == GM_NORMAL) {
1355 static auto last_time = std::chrono::steady_clock::now();
1356 auto now = std::chrono::steady_clock::now();
1357 auto delta_ms = std::chrono::duration_cast<std::chrono::milliseconds>(now - last_time);
1358 if (delta_ms.count() != 0) {
1359 TimerManager<TimerGameRealtime>::Elapsed(delta_ms);
1360 last_time = now;
1361 }
1362 }
1363
1364 /* switch game mode? */
1365 if (_switch_mode != SM_NONE && !HasModalProgress()) {
1366 SwitchToMode(_switch_mode);
1367 _switch_mode = SM_NONE;
1368 if (_exit_game) return;
1369 }
1370
1371 IncreaseSpriteLRU();
1372
1373 /* Check for UDP stuff */
1374 if (_network_available) NetworkBackgroundLoop();
1375
1376 DebugSendRemoteMessages();
1377
1378 if (_networking && !HasModalProgress()) {
1379 /* Multiplayer */
1380 NetworkGameLoop();
1381 } else {
1382 if (_network_reconnect > 0 && --_network_reconnect == 0) {
1383 /* This means that we want to reconnect to the last host
1384 * We do this here, because it means that the network is really closed */
1385 NetworkClientConnectGame(_settings_client.network.last_joined, COMPANY_SPECTATOR);
1386 }
1387 /* Singleplayer */
1388 StateGameLoop();
1389 }
1390
1391 if (_pause_mode.None() && HasBit(_display_opt, DO_FULL_ANIMATION)) DoPaletteAnimations();
