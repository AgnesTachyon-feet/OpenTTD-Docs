37 int SlopeCost(TileIndex tile, TileIndex next_tile, Trackdir)
38 {
39 /* height of the center of the current tile */
40 int x1 = TileX(tile) * TILE_SIZE;
41 int y1 = TileY(tile) * TILE_SIZE;
42 int z1 = GetSlopePixelZ(x1 + TILE_SIZE / 2, y1 + TILE_SIZE / 2, true);
43
44 /* height of the center of the next tile */
45 int x2 = TileX(next_tile) * TILE_SIZE;
46 int y2 = TileY(next_tile) * TILE_SIZE;
47 int z2 = GetSlopePixelZ(x2 + TILE_SIZE / 2, y2 + TILE_SIZE / 2, true);
48
49 if (z2 - z1 > 1) {
50 /* Slope up */
51 return Yapf().PfGetSettings().road_slope_penalty;
52 }
53 return 0;
54 }

57 inline int OneTileCost(TileIndex tile, Trackdir trackdir)
58 {
59 int cost = 0;
60 /* set base cost */
61 if (IsDiagonalTrackdir(trackdir)) {
62 cost += YAPF_TILE_LENGTH;
63 switch (GetTileType(tile)) {
64 case MP_ROAD:
65 /* Increase the cost for level crossings */
66 if (IsLevelCrossing(tile)) {
67 cost += Yapf().PfGetSettings().road_crossing_penalty;
68 }
69 break;
70
71 case MP_STATION: {
72 if (IsRoadWaypoint(tile)) break;
73
74 const RoadStop *rs = RoadStop::GetByTile(tile, GetRoadStopType(tile));
75 if (IsDriveThroughStopTile(tile)) {
76 /* Increase the cost for drive-through road stops */
77 cost += Yapf().PfGetSettings().road_stop_penalty;
78 DiagDirection dir = TrackdirToExitdir(trackdir);
79 if (!RoadStop::IsDriveThroughRoadStopContinuation(tile, tile - TileOffsByDiagDir(dir))) {
80 /* When we're the first road stop in a 'queue' of them we increase
81 * cost based on the fill percentage of the whole queue. */
82 const RoadStop::Entry &entry = rs->GetEntry(dir);
83 cost += entry.GetOccupied() * Yapf().PfGetSettings().road_stop_occupied_penalty / entry.GetLength();
84 }
85 } else {
86 /* Increase cost for filled road stops */
87 cost += Yapf().PfGetSettings().road_stop_bay_occupied_penalty * (!rs->IsFreeBay(0) + !rs->IsFreeBay(1)) / 2;
88 }
89 break;
90 }
91
92 default:
93 break;
94 }
95 } else {
96 /* non-diagonal trackdir */
97 cost = YAPF_TILE_CORNER_LENGTH + Yapf().PfGetSettings().road_curve_penalty;
98 }
99 return cost;
100 }

113 inline bool PfCalcCost(Node &n, const TrackFollower *)
114 {
115 int segment_cost = 0;
116 uint tiles = 0;
117 /* start at n.key.tile / n.key.td and walk to the end of segment */
118 TileIndex tile = n.key.tile;
119 Trackdir trackdir = n.key.td;
120 int parent_cost = (n.parent != nullptr) ? n.parent->cost : 0;
121
122 for (;;) {
123 /* base tile cost depending on distance between edges */
124 segment_cost += Yapf().OneTileCost(tile, trackdir);
125
126 const RoadVehicle *v = Yapf().GetVehicle();
127 /* we have reached the vehicle's destination - segment should end here to avoid target skipping */
128 if (Yapf().PfDetectDestinationTile(tile, trackdir)) break;
129
130 /* Finish if we already exceeded the maximum path cost (i.e. when
131 * searching for the nearest depot). */
132 if (this->max_cost > 0 && (parent_cost + segment_cost) > this->max_cost) {
133 return false;
134 }
135
136 /* stop if we have just entered the depot */
137 if (IsRoadDepotTile(tile) && trackdir == DiagDirToDiagTrackdir(ReverseDiagDir(GetRoadDepotDirection(tile)))) {
138 /* next time we will reverse and leave the depot */
139 break;
140 }
141
142 /* if there are no reachable trackdirs on new tile, we have end of road */
143 TrackFollower F(Yapf().GetVehicle());
144 if (!F.Follow(tile, trackdir)) break;
145
146 /* if there are more trackdirs available & reachable, we are at the end of segment */
147 if (KillFirstBit(F.new_td_bits) != TRACKDIR_BIT_NONE) break;
148
149 Trackdir new_td = (Trackdir)FindFirstBit(F.new_td_bits);
150
151 /* stop if RV is on simple loop with no junctions */
152 if (F.new_tile == n.key.tile && new_td == n.key.td) return false;
153
154 /* if we skipped some tunnel tiles, add their cost */
155 segment_cost += F.tiles_skipped * YAPF_TILE_LENGTH;
156 tiles += F.tiles_skipped + 1;
157
158 /* add hilly terrain penalty */
159 segment_cost += Yapf().SlopeCost(tile, F.new_tile, trackdir);
160
161 /* add min/max speed penalties */
162 int min_speed = 0;
163 int max_veh_speed = std::min<int>(v->GetDisplayMaxSpeed(), v->current_order.GetMaxSpeed() * 2);
164 int max_speed = F.GetSpeedLimit(&min_speed);
165 if (max_speed < max_veh_speed) segment_cost += YAPF_TILE_LENGTH * (max_veh_speed - max_speed) * (4 + F.tiles_skipped) / max_veh_speed;
166 if (min_speed > max_veh_speed) segment_cost += YAPF_TILE_LENGTH * (min_speed - max_veh_speed);
167
168 /* move to the next tile */
169 tile = F.new_tile;
170 trackdir = new_td;
171 if (tiles > MAX_MAP_SIZE) break;
172 }
173
174 /* save end of segment back to the node */
175 n.segment_last_tile = tile;
176 n.segment_last_td = trackdir;
177
178 /* save also tile cost */
179 n.cost = parent_cost + segment_cost;
180 return true;
181 }

238 void SetDestination(const RoadVehicle *v)
239 {
240 if (v->current_order.IsType(OT_GOTO_STATION)) {
241 this->dest_station = v->current_order.GetDestination().ToStationID();
242 this->station_type = v->IsBus() ? StationType::Bus : StationType::Truck;

244 this->non_artic = !v->HasArticulatedPart();
245 this->dest_trackdirs = INVALID_TRACKDIR_BIT;
246 } else if (v->current_order.IsType(OT_GOTO_WAYPOINT)) {
247 this->dest_station = v->current_order.GetDestination().ToStationID();
248 this->station_type = StationType::RoadWaypoint;
249 this->dest_tile = CalcClosestStationTile(this->dest_station, v->tile, this->station_type);
250 this->non_artic = !v->HasArticulatedPart();
251 this->dest_trackdirs = INVALID_TRACKDIR_BIT;
252 } else {
253 this->dest_station = StationID::Invalid();
254 this->dest_tile = v->dest_tile;
255 this->dest_trackdirs = TrackStatusToTrackdirBits(GetTileTrackStatus(v->dest_tile, TRANSPORT_ROAD, GetRoadTramType(v->roadtype)));
256 }

278 inline bool PfDetectDestinationTile(TileIndex tile, Trackdir trackdir)
279 {
280 if (this->dest_station != StationID::Invalid()) {
281 return IsTileType(tile, MP_STATION) &&
282 GetStationIndex(tile) == this->dest_station &&
283 (this->station_type == GetStationType(tile)) &&
284 (this->non_artic || IsDriveThroughStopTile(tile));
285 }
286
287 return tile == this->dest_tile && HasTrackdir(this->dest_trackdirs, trackdir);
288 }

294 inline bool PfCalcEstimate(Node &n)
295 {
296 if (this->PfDetectDestination(n)) {
297 n.estimate = n.cost;
298 return true;
299 }
300
301 n.estimate = n.cost + OctileDistanceCost(n.segment_last_tile, n.segment_last_td, this->dest_tile);
302 assert(n.estimate >= n.parent->estimate);
303 return true;
304 }

351 inline Trackdir ChooseRoadTrack(const RoadVehicle *v, TileIndex tile, DiagDirection enterdir, bool &path_found, RoadVehPathCache &path_cache)
352 {
353 /* Handle special case - when next tile is destination tile.
354 * However, when going to a station the (initial) destination
355 * tile might not be a station, but a junction, in which case
356 * this method forces the vehicle to jump in circles. */
357 if (tile == v->dest_tile && !v->current_order.IsType(OT_GOTO_STATION)) {
358 /* choose diagonal trackdir reachable from enterdir */
359 return DiagDirToDiagTrackdir(enterdir);
360 }
361 /* our source tile will be the next vehicle tile (should be the given one) */
362 TileIndex src_tile = tile;
363 /* get available trackdirs on the start tile */
364 TrackdirBits src_trackdirs = GetTrackdirBitsForRoad(tile, GetRoadTramType(v->roadtype));
365 /* select reachable trackdirs only */
366 src_trackdirs &= DiagdirReachesTrackdirs(enterdir);
367
368 /* set origin and destination nodes */
369 Yapf().SetOrigin(src_tile, src_trackdirs);
370 Yapf().SetDestination(v);
371
372 /* find the best path */
373 path_found = Yapf().FindPath(v);
374
375 /* if path not found - return INVALID_TRACKDIR */
376 Trackdir next_trackdir = INVALID_TRACKDIR;
377 Node *node = Yapf().GetBestNode();
378 if (node != nullptr) {
379 uint steps = 0;
380 for (Node *n = node; n->parent != nullptr; n = n->parent) steps++;
381
382 /* path was found or at least suggested
383 * walk through the path back to its origin */
384 while (node->parent != nullptr) {
385 steps--;
386 if (node->GetIsChoice() && steps < YAPF_ROADVEH_PATH_CACHE_SEGMENTS) {
387 path_cache.emplace_back(node->GetTrackdir(), node->GetTile());
388 }
389 node = node->parent;
390 }
391 /* return trackdir from the best origin node (one of start nodes) */
392 Node &best_next_node = *node;
393 assert(best_next_node.GetTile() == tile);
394 next_trackdir = best_next_node.GetTrackdir();
395
396 /* Check if target is a station, and cached path leads to within YAPF_ROADVEH_PATH_CACHE_DESTINATION_LIMIT
397 * tiles of the dest tile */
398 const Station *st = Yapf().GetDestinationStation();
399 if (st) {
400 const RoadStop *stop = st->GetPrimaryRoadStop(v);
401 if (stop != nullptr && (IsDriveThroughStopTile(stop->xy) || stop->GetNextRoadStop(v) != nullptr)) {
402 /* Destination station has at least 2 usable road stops, or first is a drive-through stop,
403 * trim end of path cache within a number of tiles of road stop tile area */
404 TileArea non_cached_area = v->IsBus() ? st->bus_station : st->truck_station;
405 non_cached_area.Expand(YAPF_ROADVEH_PATH_CACHE_DESTINATION_LIMIT);
406
407 /* Find the first tile not contained by the non-cacheable area, and remove from the cache. */
408 auto it = std::find_if(std::begin(path_cache), std::end(path_cache), [&non_cached_area](const auto &pc) { return !non_cached_area.Contains(pc.tile); });
409 path_cache.erase(std::begin(path_cache), it);
410 }
411 }
412 }
413 return next_trackdir;
414 }

416 inline uint DistanceToTile(const RoadVehicle *v, TileIndex dst_tile)
417 {
418 /* handle special case - when current tile is the destination tile */
419 if (dst_tile == v->tile) {
420 /* distance is zero in this case */
421 return 0;
422 }
423
424 if (!this->SetOriginFromVehiclePos(v)) return UINT_MAX;
425
426 /* get available trackdirs on the destination tile */
427 Yapf().SetDestination(v);
428
429 /* if path not found - return distance = UINT_MAX */
430 uint dist = UINT_MAX;
431
432 /* find the best path */
433 if (!Yapf().FindPath(v)) return dist;
434
435 Node *node = Yapf().GetBestNode();
436 if (node != nullptr) {
437 /* path was found
438 * get the path cost estimate */
439 dist = node->GetCostEstimate();
440 }
441
442 return dist;
443 }

446 inline bool SetOriginFromVehiclePos(const RoadVehicle *v)
447 {
448 /* set origin (tile, trackdir) */
449 TileIndex src_tile = v->tile;
450 Trackdir src_td = v->GetVehicleTrackdir();
451 if (!HasTrackdir(GetTrackdirBitsForRoad(src_tile, Yapf().IsTram() ? RTT_TRAM : RTT_ROAD), src_td)) {
452 /* sometimes the roadveh is not on the road (it resides on non-existing track)
453 * how should we handle that situation? */
454 return false;
455 }
456 Yapf().SetOrigin(src_tile, TrackdirToTrackdirBits(src_td));
457 return true;
458 }

473 inline FindDepotData FindNearestDepot(const RoadVehicle *v, TileIndex tile, Trackdir td, int max_distance)
474 {
475 /* Set origin. */
476 Yapf().SetOrigin(tile, TrackdirToTrackdirBits(td));
477 Yapf().SetMaxCost(max_distance);
478
479 /* Find the best path and return if no depot is found. */
480 if (!Yapf().FindPath(v)) return FindDepotData();

483 Node *n = Yapf().GetBestNode();
