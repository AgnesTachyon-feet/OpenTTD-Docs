58inline CargoMonitorID EncodeCargoIndustryMonitor(CompanyID company, CargoType ctype, IndustryID ind)
59{
60 assert(ctype < (1 << CCB_CARGO_TYPE_LENGTH));
61 assert(company < (1 << CCB_COMPANY_LENGTH));
62
63 uint32_t ret = 0;
64 SB(ret, CCB_TOWN_IND_NUMBER_START, CCB_TOWN_IND_NUMBER_LENGTH, ind.base());
65 SetBit(ret, CCB_IS_INDUSTRY_BIT);
66 SB(ret, CCB_CARGO_TYPE_START, CCB_CARGO_TYPE_LENGTH, ctype);
67 SB(ret, CCB_COMPANY_START, CCB_COMPANY_LENGTH, company.base());
68 return ret;
69}
