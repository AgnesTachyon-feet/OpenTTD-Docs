93inline void ToggleInvisibilityWithTransparency(TransparencyOption to)
94{
95 if (IsInvisibilitySet(to)) {
96 ClrBit(_invisibility_opt, to);
97 ClrBit(_transparency_opt, to);
98 } else {
99 SetBit(_invisibility_opt, to);
100 SetBit(_transparency_opt, to);
101 }
102}

115inline void ResetRestoreAllTransparency()
116{
117 /* if none of the non-locked options are set */
118 if ((_transparency_opt & ~_transparency_lock) == 0) {
119 /* set all non-locked options */
120 _transparency_opt |= GB(~_transparency_lock, 0, TO_END);
121 } else {
122 /* clear all non-locked options */
123 _transparency_opt &= _transparency_lock;
124 }
125
126 MarkWholeScreenDirty();
127}
