20static void LoadFontGlyph(ByteReader &buf)
21{
22 /* <12> <num_def> <font_size> <num_char> <base_char>
23 *
24 * B num_def Number of definitions
25 * B font_size Size of font (0 = normal, 1 = small, 2 = large, 3 = mono)
26 * B num_char Number of consecutive glyphs
27 * W base_char First character index */
28
29 uint8_t num_def = buf.ReadByte();
30
31 for (uint i = 0; i < num_def; i++) {
32 FontSize size = (FontSize)buf.ReadByte();
33 uint8_t num_char = buf.ReadByte();
34 uint16_t base_char = buf.ReadWord();
35
36 if (size >= FS_END) {
37 GrfMsg(1, "LoadFontGlyph: Size {} is not supported, ignoring", size);
38 }
39
40 GrfMsg(7, "LoadFontGlyph: Loading {} glyph(s) at 0x{:04X} for size {}", num_char, base_char, size);
41
42 for (uint c = 0; c < num_char; c++) {
43 if (size < FS_END) SetUnicodeGlyph(size, base_char + c, _cur_gps.spriteid);
44 _cur_gps.nfo_line++;
45 LoadNextSprite(_cur_gps.spriteid++, *_cur_gps.file, _cur_gps.nfo_line);
46 }
47 }
48}

51static void SkipAct12(ByteReader &buf)
52{
53 /* <12> <num_def> <font_size> <num_char> <base_char>
54 *
55 * B num_def Number of definitions
56 * B font_size Size of font (0 = normal, 1 = small, 2 = large)
57 * B num_char Number of consecutive glyphs
58 * W base_char First character index */
59
60 uint8_t num_def = buf.ReadByte();
61
62 for (uint i = 0; i < num_def; i++) {
63 /* Ignore 'size' byte */
64 buf.ReadByte();
65
66 /* Sum up number of characters */
67 _cur_gps.skip_sprites += buf.ReadByte();
68
69 /* Ignore 'base_char' word */
70 buf.ReadWord();
71 }
72
73 GrfMsg(3, "SkipAct12: Skipping {} sprites", _cur_gps.skip_sprites);
74}
