39/* SPRITE_WIDTH is 24. ECS has roughly 30 sprite groups per real sprite.
40 * Adding an 'extra' margin would be assuming 64 sprite groups per real
41 * sprite. 64 = 2^6, so 2^30 should be enough (for now) */

288struct ResolverObject {
289private:
290 static TemporaryStorageArray<int32_t, 0x110> temp_store;
291
292public:
300 ResolverObject(const GRFFile *grffile, CallbackID callback = CBID_NO_CALLBACK, uint32_t callback_param1 = 0, uint32_t callback_param2 = 0)
301 : default_scope(*this), callback(callback), callback_param1(callback_param1), callback_param2(callback_param2), grffile(grffile), root_spritegroup(nullptr)
302 {
303 }
304
305 virtual ~ResolverObject() = default;
306
307 ResolverResult DoResolve()
308 {
309 temp_store.ClearChanges();
310 this->last_value = 0;
311 this->used_random_triggers = 0;
312 this->reseed.fill(0);
313 return SpriteGroup::Resolve(this->root_spritegroup, *this);
314 }
315
316 ScopeResolver default_scope;
317
324 inline int32_t GetRegister(uint i) const
325 {
326 return temp_store.GetValue(i);
327 }
328
335 inline void SetRegister(uint i, int32_t value)
336 {
337 temp_store.StoreValue(i, value);
338 }
339
340 CallbackID callback{};
341 uint32_t callback_param1 = 0;
342 uint32_t callback_param2 = 0;
343
344 uint32_t last_value = 0;
345
346protected:
347 uint32_t waiting_random_triggers = 0;
348 uint32_t used_random_triggers = 0;
349public:
350 std::array<uint32_t, VSG_END> reseed;
351
352 const GRFFile *grffile = nullptr;
353 const SpriteGroup *root_spritegroup = nullptr;
354
360 template <class TSpriteGroup>
361 inline const TSpriteGroup *Resolve()
362 {
363 auto result = this->DoResolve();
364 const auto *group = std::get_if<const TSpriteGroup *>(&result);
365 if (group == nullptr) return nullptr;
366 return *group;
367 }
368
376 inline void ResolveRerandomisation()
377 {
378 /* The Resolve result has no meaning.
379 * It can be a SpriteSet, a callback result, or even an invalid SpriteGroup reference (nullptr). */
380 this->DoResolve();
381 }
382
388 inline CallbackResult ResolveCallback(std::span<int32_t> regs100)
389 {
390 auto result = this->DoResolve();
391 const auto *value = std::get_if<CallbackResult>(&result);
392 if (value == nullptr) return CALLBACK_FAILED;
393 for (uint i = 0; i < regs100.size(); ++i) {
394 regs100[i] = this->GetRegister(0x100 + i);
395 }
396 return *value;
397 }
398
399 virtual const SpriteGroup *ResolveReal(const RealSpriteGroup &group) const;
400
401 virtual ScopeResolver *GetScope(VarSpriteGroupScope scope = VSG_SCOPE_SELF, uint8_t relative = 0);
402
406 uint32_t GetWaitingRandomTriggers() const
407 {
408 return this->waiting_random_triggers;
409 }
410
414 void AddUsedRandomTriggers(uint32_t triggers)
415 {
416 this->used_random_triggers |= triggers;
417 }
418
424 uint32_t GetReseedSum() const
425 {
426 uint32_t sum = 0;
427 for (VarSpriteGroupScope vsg = VSG_BEGIN; vsg < VSG_END; vsg++) {
428 sum |= this->reseed[vsg];
429 }
430 return sum;
431 }
432
437 virtual GrfSpecFeature GetFeature() const { return GSF_INVALID; }
443 virtual uint32_t GetDebugID() const { return 0; }
444};
