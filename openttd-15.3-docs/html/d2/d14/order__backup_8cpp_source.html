37OrderBackup::OrderBackup(const Vehicle *v, uint32_t user) : user(user), tile(v->tile), group(v->group_id)
38{
39 this->CopyConsistPropertiesFrom(v);
40
41 /* If we have shared orders, store the vehicle we share the order with. */
42 if (v->IsOrderListShared()) {
43 this->clone = (v->FirstShared() == v) ? v->NextShared() : v->FirstShared();
44 } else {
45 /* Else copy the orders */
46 this->orders.assign(std::begin(v->Orders()), std::end(v->Orders()));
47 }
48}

54void OrderBackup::DoRestore(Vehicle *v)
55{
56 /* If we had shared orders, recover that */
57 if (this->clone != nullptr) {
58 Command<CMD_CLONE_ORDER>::Do(DoCommandFlag::Execute, CO_SHARE, v->index, this->clone->index);
59 } else if (!this->orders.empty() && OrderList::CanAllocateItem()) {
60 v->orders = new OrderList(std::move(this->orders), v);
61 /* Make sure buoys/oil rigs are updated in the station list. */
62 InvalidateWindowClassesData(WC_STATION_LIST, 0);
63 }
64
65 /* Remove backed up name if it's no longer unique. */
66 if (!IsUniqueVehicleName(this->name)) this->name.clear();
67
68 v->CopyConsistPropertiesFrom(this);
69
70 /* Make sure orders are in range */
71 v->UpdateRealOrderIndex();
72 if (v->cur_implicit_order_index >= v->GetNumOrders()) v->cur_implicit_order_index = v->cur_real_order_index;
73
74 /* Restore vehicle group */
75 Command<CMD_ADD_VEHICLE_GROUP>::Do(DoCommandFlag::Execute, this->group, v->index, false, VehicleListIdentifier{});
76}

84/* static */ void OrderBackup::Backup(const Vehicle *v, uint32_t user)
85{
86 /* Don't use reset as that broadcasts over the network to reset the variable,
87 * which is what we are doing at the moment. */
88 for (OrderBackup *ob : OrderBackup::Iterate()) {
89 if (ob->user == user) delete ob;
90 }
91 if (OrderBackup::CanAllocateItem()) {
92 new OrderBackup(v, user);
93 }
94}

102/* static */ void OrderBackup::Restore(Vehicle *v, uint32_t user)
103{

146/* static */ void OrderBackup::ResetUser(uint32_t user)
147{
148 assert(_network_server);

150 for (OrderBackup *ob : OrderBackup::Iterate()) {
151 /* If it's not a backup of us, ignore it. */
152 if (ob->user != user) continue;

165/* static */ void OrderBackup::Reset(TileIndex t, bool from_gui)
166{
167 /* The user has CLIENT_ID_SERVER as default when network play is not active,
168 * but compiled it. A network client has its own variable for the unique
169 * client/user identifier. Finally if networking isn't compiled in the
170 * default is just plain and simple: 0. */
171 uint32_t user = _networking && !_network_server ? _network_own_client_id : CLIENT_ID_SERVER;
172
173 for (OrderBackup *ob : OrderBackup::Iterate()) {
174 /* If this is a GUI action, and it's not a backup of us, ignore it. */
175 if (from_gui && ob->user != user) continue;
176 /* If it's not for our chosen tile either, ignore it. */
177 if (t != INVALID_TILE && t != ob->tile) continue;
178
179 if (from_gui) {
180 /* We need to circumvent the "prevention" from this command being executed
181 * while the game is paused, so use the internal method. Nor do we want
182 * this command to get its cost estimated when shift is pressed. */
183 Command<CMD_CLEAR_ORDER_BACKUP>::Unsafe<CommandCallback>(STR_NULL, nullptr, true, false, ob->tile, CommandTraits<CMD_CLEAR_ORDER_BACKUP>::Args{ ob->tile, static_cast<ClientID>(user) });
184 } else {
185 /* The command came from the game logic, i.e. the clearing of a tile.
186 * In that case we have no need to actually sync this, just do it. */
187 delete ob;
188 }
189 }
190}

210/* static */ void OrderBackup::ClearVehicle(const Vehicle *v)
211{
212 assert(v != nullptr);
213 for (OrderBackup *ob : OrderBackup::Iterate()) {
214 if (ob->clone == v) {
215 /* Get another item in the shared list. */
216 ob->clone = (v->FirstShared() == v) ? v->NextShared() : v->FirstShared();
217 /* But if that isn't there, remove it. */
218 if (ob->clone == nullptr) delete ob;
219 }
220 }

231/* static */ void OrderBackup::RemoveOrder(OrderType type, DestinationID destination, bool hangar)
232{
233 for (OrderBackup *ob : OrderBackup::Iterate()) {
234 for (Order &order : ob->orders) {
235 OrderType ot = order.GetType();
236 if (ot == OT_GOTO_DEPOT && order.GetDepotActionType().Test(OrderDepotActionFlag::NearestDepot)) continue;
237 if (ot == OT_GOTO_DEPOT && hangar && !IsHangarTile(ob->tile)) continue; // Not an aircraft? Can't have a hangar order.
238 if (ot == OT_IMPLICIT || (IsHangarTile(ob->tile) && ot == OT_GOTO_DEPOT && !hangar)) ot = OT_GOTO_STATION;
239 if (ot == type && order.GetDestination() == destination) {
240 /* Remove the order backup! If a station/depot gets removed, we can't/shouldn't restore those broken orders. */
241 delete ob;
242 break;
243 }
244 }
245 }
