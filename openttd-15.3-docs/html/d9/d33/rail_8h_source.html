410inline bool Rail90DegTurnDisallowed(RailType rt1, RailType rt2, bool def = _settings_game.pf.forbid_90_deg)
411{
412 if (rt1 == INVALID_RAILTYPE || rt2 == INVALID_RAILTYPE) return def;
413
414 const RailTypeInfo *rti1 = GetRailTypeInfo(rt1);
415 const RailTypeInfo *rti2 = GetRailTypeInfo(rt2);
416
417 bool rt1_90deg = rti1->flags.Test(RailTypeFlag::Disallow90Deg) || (!rti1->flags.Test(RailTypeFlag::Allow90Deg) && def);
418 bool rt2_90deg = rti2->flags.Test(RailTypeFlag::Disallow90Deg) || (!rti2->flags.Test(RailTypeFlag::Allow90Deg) && def);
419
420 return rt1_90deg || rt2_90deg;
421}

428inline Money RailBuildCost(RailType railtype)
429{
430 assert(railtype < RAILTYPE_END);
431 return (_price[PR_BUILD_RAIL] * GetRailTypeInfo(railtype)->cost_multiplier) >> 3;
432}

439inline Money RailClearCost(RailType railtype)
440{
441 /* Clearing rail in fact earns money, but if the build cost is set
442 * very low then a loophole exists where money can be made.
443 * In this case we limit the removal earnings to 3/4s of the build
444 * cost.
445 */
446 assert(railtype < RAILTYPE_END);
447 return std::max(_price[PR_CLEAR_RAIL], -RailBuildCost(railtype) * 3 / 4);
448}
449
456inline Money RailConvertCost(RailType from, RailType to)
457{
458 /* Get the costs for removing and building anew
459 * A conversion can never be more costly */
460 Money rebuildcost = RailBuildCost(to) + RailClearCost(from);
461
462 /* Conversion between somewhat compatible railtypes:
463 * Pay 1/8 of the target rail cost (labour costs) and additionally any difference in the
464 * build costs, if the target type is more expensive (material upgrade costs).
465 * Upgrade can never be more expensive than re-building. */
466 if (HasPowerOnRail(from, to) || HasPowerOnRail(to, from)) {
467 Money upgradecost = RailBuildCost(to) / 8 + std::max((Money)0, RailBuildCost(to) - RailBuildCost(from));
468 return std::min(upgradecost, rebuildcost);
469 }
470
471 /* make the price the same as remove + build new type for rail types
472 * which are not compatible in any way */
473 return rebuildcost;
474}

483inline Money RailMaintenanceCost(RailType railtype, uint32_t num, uint32_t total_num)
484{
485 assert(railtype < RAILTYPE_END);
486 return (_price[PR_INFRASTRUCTURE_RAIL] * GetRailTypeInfo(railtype)->maintenance_multiplier * num * (1 + IntSqrt(total_num))) >> 11; // 4 bits fraction for the multiplier and 7 bits scaling.
487}
488
494inline Money SignalMaintenanceCost(uint32_t num)
495{
496 return (_price[PR_INFRASTRUCTURE_RAIL] * 15 * num * (1 + IntSqrt(num))) >> 8; // 1 bit fraction for the multiplier and 7 bits scaling.
497}
