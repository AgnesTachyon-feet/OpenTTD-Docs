58bool ClientNetworkTurnSocketHandler::Receive_TURN_CONNECTED(Packet &p)
59{
60 std::string hostname = p.Recv_string(NETWORK_HOSTNAME_LENGTH);
61 Debug(net, 9, "Turn::Receive_TURN_CONNECTED({})", hostname);
62
63 /* Act like we no longer have a socket, as we are handing it over to the
64 * game handler. */
65 SOCKET game_sock = this->sock;
66 this->sock = INVALID_SOCKET;
67
68 NetworkAddress address = NetworkAddress(hostname, NETWORK_DEFAULT_PORT);
69 _network_coordinator_client.ConnectSuccess(this->token, game_sock, address);
70
71 return false;
72}

95/* static */ std::unique_ptr<ClientNetworkTurnSocketHandler> ClientNetworkTurnSocketHandler::Turn(std::string_view token, uint8_t tracking_number, std::string_view ticket, std::string_view connection_string)
96{
97 auto turn_handler = std::make_unique<ClientNetworkTurnSocketHandler>(token, tracking_number, connection_string);
98
99 auto p = std::make_unique<Packet>(turn_handler.get(), PACKET_TURN_SERCLI_CONNECT);
100 p->Send_uint8(NETWORK_COORDINATOR_VERSION);
101 p->Send_string(ticket);
102
103 Debug(net, 9, "Turn::SendTurn({})", ticket);
104 turn_handler->SendPacket(std::move(p));
105
106 return turn_handler;
107}
