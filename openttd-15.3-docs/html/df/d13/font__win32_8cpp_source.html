50static int CALLBACK EnumFontCallback(const ENUMLOGFONTEX *logfont, const NEWTEXTMETRICEX *metric, DWORD type, LPARAM lParam)
51{
52 EFCParam *info = (EFCParam *)lParam;
53
54 /* Skip duplicates */
55 if (!info->Add(logfont->elfFullName)) return 1;
56 /* Only use TrueType fonts */
57 if (!(type & TRUETYPE_FONTTYPE)) return 1;
58 /* Don't use SYMBOL fonts */
59 if (logfont->elfLogFont.lfCharSet == SYMBOL_CHARSET) return 1;
60 /* Use monospaced fonts when asked for it. */
61 if (info->callback->Monospace() && (logfont->elfLogFont.lfPitchAndFamily & (FF_MODERN | FIXED_PITCH)) != (FF_MODERN | FIXED_PITCH)) return 1;
62
63 /* The font has to have at least one of the supported locales to be usable. */
64 auto check_bitfields = [&]() {
65 /* First try Unicode Subset Bitfield. */
66 for (uint8_t i = 0; i < 4; i++) {
67 if ((metric->ntmFontSig.fsUsb[i] & info->locale.lsUsb[i]) != 0) return true;
68 }
69 /* Keep Code Page Bitfield as a fallback. */
70 for (uint8_t i = 0; i < 2; i++) {
71 if ((metric->ntmFontSig.fsCsb[i] & info->locale.lsCsbSupported[i]) != 0) return true;
72 }
73 return false;
74 };
75 if (!check_bitfields()) return 1;
76
77 char font_name[MAX_PATH];
78 convert_from_fs(logfont->elfFullName, font_name);
79
80 info->callback->SetFontNames(info->settings, font_name, &logfont->elfLogFont);
81 if (info->callback->FindMissingGlyphs()) return 1;
82 Debug(fontcache, 1, "Fallback font: {}", font_name);
83 return 0; // stop enumerating
84}

105void Win32FontCache::SetFontSize(int pixels)
106{
107 if (pixels == 0) {
108 /* Try to determine a good height based on the minimal height recommended by the font. */
109 int scaled_height = ScaleGUITrad(FontCache::GetDefaultFontHeight(this->fs));
110 pixels = scaled_height;
111
112 HFONT temp = CreateFontIndirect(&this->logfont);
113 if (temp != nullptr) {
114 HGDIOBJ old = SelectObject(this->dc, temp);
115
116 UINT size = GetOutlineTextMetrics(this->dc, 0, nullptr);
117 LPOUTLINETEXTMETRIC otm = (LPOUTLINETEXTMETRIC)new BYTE[size];
118 GetOutlineTextMetrics(this->dc, size, otm);
119
120 /* Font height is minimum height plus the difference between the default
121 * height for this font size and the small size. */
122 int diff = scaled_height - ScaleGUITrad(FontCache::GetDefaultFontHeight(FS_SMALL));
123 /* Clamp() is not used as scaled_height could be greater than MAX_FONT_SIZE, which is not permitted in Clamp(). */
124 pixels = std::min(std::max(std::min<int>(otm->otmusMinimumPPEM, MAX_FONT_MIN_REC_SIZE) + diff, scaled_height), MAX_FONT_SIZE);
125
126 delete[] (BYTE*)otm;
127 SelectObject(dc, old);
128 DeleteObject(temp);
129 }
130 } else {
131 pixels = ScaleGUITrad(pixels);
132 }
133 this->used_size = pixels;
134
135 /* Create GDI font handle. */
136 this->logfont.lfHeight = -pixels;
137 this->logfont.lfWidth = 0;
138 this->logfont.lfOutPrecision = OUT_TT_ONLY_PRECIS;
139 this->logfont.lfQuality = ANTIALIASED_QUALITY;
140
141 if (this->font != nullptr) {
142 SelectObject(dc, this->old_font);
143 DeleteObject(this->font);
144 }
145 this->font = CreateFontIndirect(&this->logfont);
146 this->old_font = SelectObject(this->dc, this->font);
147
148 /* Query the font metrics we needed. */
149 UINT otmSize = GetOutlineTextMetrics(this->dc, 0, nullptr);
150 POUTLINETEXTMETRIC otm = (POUTLINETEXTMETRIC)new BYTE[otmSize];
151 GetOutlineTextMetrics(this->dc, otmSize, otm);
152
153 this->ascender = otm->otmTextMetrics.tmAscent;
154 this->descender = otm->otmTextMetrics.tmDescent;
155 this->height = this->ascender + this->descender;
156 this->glyph_size.cx = otm->otmTextMetrics.tmMaxCharWidth;
157 this->glyph_size.cy = otm->otmTextMetrics.tmHeight;
158
159 this->fontname = FS2OTTD((LPWSTR)((BYTE *)otm + (ptrdiff_t)otm->otmpFaceName));
160
161 Debug(fontcache, 2, "Loaded font '{}' with size {}", this->fontname, pixels);
162 delete[] (BYTE*)otm;
163}

176/* virtual */ const Sprite *Win32FontCache::InternalGetGlyph(GlyphID key, bool aa)
177{
178 GLYPHMETRICS gm;
179 MAT2 mat = { {0, 1}, {0, 0}, {0, 0}, {0, 1} };
180
181 /* Call GetGlyphOutline with zero size initially to get required memory size. */
182 DWORD size = GetGlyphOutline(this->dc, key, GGO_GLYPH_INDEX | (aa ? GGO_GRAY8_BITMAP : GGO_BITMAP), &gm, 0, nullptr, &mat);
183 if (size == GDI_ERROR) UserError("Unable to render font glyph");
184
185 /* Add 1 scaled pixel for the shadow on the medium font. Our sprite must be at least 1x1 pixel. */
186 uint shadow = (this->fs == FS_NORMAL) ? ScaleGUITrad(1) : 0;
187 uint width = std::max(1U, (uint)gm.gmBlackBoxX + shadow);
188 uint height = std::max(1U, (uint)gm.gmBlackBoxY + shadow);
189
190 /* Limit glyph size to prevent overflows later on. */
191 if (width > MAX_GLYPH_DIM || height > MAX_GLYPH_DIM) UserError("Font glyph is too large");
192
193 /* Call GetGlyphOutline again with size to actually render the glyph. */
194 uint8_t *bmp = this->render_buffer.Allocate(size);
195 GetGlyphOutline(this->dc, key, GGO_GLYPH_INDEX | (aa ? GGO_GRAY8_BITMAP : GGO_BITMAP), &gm, size, bmp, &mat);
196
197 /* GDI has rendered the glyph, now we allocate a sprite and copy the image into it. */
198 SpriteLoader::SpriteCollection spritecollection;
199 SpriteLoader::Sprite &sprite = spritecollection[ZoomLevel::Min];
200 sprite.AllocateData(ZoomLevel::Min, width * height);
201 sprite.colours = SpriteComponent::Palette;
202 if (aa) sprite.colours.Set(SpriteComponent::Alpha);
203 sprite.width = width;
204 sprite.height = height;
205 sprite.x_offs = gm.gmptGlyphOrigin.x;
206 sprite.y_offs = this->ascender - gm.gmptGlyphOrigin.y;
207
208 if (size > 0) {
209 /* All pixel data returned by GDI is in the form of DWORD-aligned rows.
210 * For a non anti-aliased glyph, the returned bitmap has one bit per pixel.
211 * For anti-aliased rendering, GDI uses the strange value range of 0 to 64,
212 * inclusively. To map this to 0 to 255, we shift left by two and then
213 * subtract one. */
214 uint pitch = Align(aa ? gm.gmBlackBoxX : std::max((gm.gmBlackBoxX + 7u) / 8u, 1u), 4);
215
216 /* Draw shadow for medium size. */
217 if (this->fs == FS_NORMAL && !aa) {
218 for (uint y = 0; y < gm.gmBlackBoxY; y++) {
219 for (uint x = 0; x < gm.gmBlackBoxX; x++) {
220 if (aa ? (bmp[x + y * pitch] > 0) : HasBit(bmp[(x / 8) + y * pitch], 7 - (x % 8))) {
221 sprite.data[shadow + x + (shadow + y) * sprite.width].m = SHADOW_COLOUR;
222 sprite.data[shadow + x + (shadow + y) * sprite.width].a = aa ? (bmp[x + y * pitch] << 2) - 1 : 0xFF;
223 }
224 }
225 }
226 }
227
228 for (uint y = 0; y < gm.gmBlackBoxY; y++) {
229 for (uint x = 0; x < gm.gmBlackBoxX; x++) {
230 if (aa ? (bmp[x + y * pitch] > 0) : HasBit(bmp[(x / 8) + y * pitch], 7 - (x % 8))) {
231 sprite.data[x + y * sprite.width].m = FACE_COLOUR;
232 sprite.data[x + y * sprite.width].a = aa ? (bmp[x + y * pitch] << 2) - 1 : 0xFF;
233 }
234 }
235 }
236 }
237
238 UniquePtrSpriteAllocator allocator;
239 BlitterFactory::GetCurrentBlitter()->Encode(SpriteType::Font, spritecollection, allocator);
240
241 GlyphEntry new_glyph;
242 new_glyph.data = std::move(allocator.data);
243 new_glyph.width = gm.gmCellIncX;
244
245 return this->SetGlyphPtr(key, std::move(new_glyph)).GetSprite();
246}

248/* virtual */ GlyphID Win32FontCache::MapCharToGlyph(char32_t key, bool allow_fallback)
249{
250 assert(IsPrintable(key));
251
252 /* Convert characters outside of the BMP into surrogate pairs. */
253 WCHAR chars[2];
254 if (key >= 0x010000U) {
255 chars[0] = (wchar_t)(((key - 0x010000U) >> 10) + 0xD800);
256 chars[1] = (wchar_t)(((key - 0x010000U) & 0x3FF) + 0xDC00);
257 } else {
258 chars[0] = (wchar_t)(key & 0xFFFF);
259 }
260
261 WORD glyphs[2] = { 0, 0 };
262 GetGlyphIndicesW(this->dc, chars, key >= 0x010000U ? 2 : 1, glyphs, GGI_MARK_NONEXISTING_GLYPHS);
263
264 if (glyphs[0] != 0xFFFF) return glyphs[0];
265 return allow_fallback && key >= SCC_SPRITE_START && key <= SCC_SPRITE_END ? this->parent->MapCharToGlyph(key) : 0;
266}

278 std::unique_ptr<FontCache> LoadFont(FontSize fs, FontType fonttype) const override
279 {
280 if (fonttype != FontType::TrueType) return nullptr;
281
282 FontCacheSubSetting *settings = GetFontCacheSubSetting(fs);
283
284 std::string font = GetFontCacheFontName(fs);
285 if (font.empty()) return nullptr;
286
287 LOGFONT logfont{};
288 logfont.lfPitchAndFamily = fs == FS_MONO ? FIXED_PITCH : VARIABLE_PITCH;
289 logfont.lfCharSet = DEFAULT_CHARSET;
290 logfont.lfOutPrecision = OUT_OUTLINE_PRECIS;
291 logfont.lfClipPrecision = CLIP_DEFAULT_PRECIS;
292
293 if (settings->os_handle != nullptr) {
294 logfont = *(const LOGFONT *)settings->os_handle;
295 } else if (font.find('.') != std::string::npos) {
296 /* Might be a font file name, try load it. */
297 if (!TryLoadFontFromFile(font, logfont)) {
298 ShowInfo("Unable to load file '{}' for {} font, using default windows font selection instead", font, FontSizeToName(fs));
299 }
300 }
301
302 if (logfont.lfFaceName[0] == 0) {
303 logfont.lfWeight = StrContainsIgnoreCase(font, " bold") ? FW_BOLD : FW_NORMAL; // Poor man's way to allow selecting bold fonts.
304 convert_to_fs(font, logfont.lfFaceName);
305 }
306
307 return LoadWin32Font(fs, logfont, GetFontCacheFontSize(fs), font);
308 }

310 bool FindFallbackFont(FontCacheSettings *settings, const std::string &language_isocode, MissingGlyphSearcher *callback) const override
311 {
312 Debug(fontcache, 1, "Trying fallback fonts");
313 EFCParam langInfo;
314 std::wstring lang = OTTD2FS(language_isocode.substr(0, language_isocode.find('_')));
315 if (GetLocaleInfoEx(lang.c_str(), LOCALE_FONTSIGNATURE, reinterpret_cast<LPWSTR>(&langInfo.locale), sizeof(langInfo.locale) / sizeof(wchar_t)) == 0) {
316 /* Invalid isocode or some other mysterious error, can't determine fallback font. */
317 Debug(fontcache, 1, "Can't get locale info for fallback font (isocode={})", language_isocode);
318 return false;
319 }
320 langInfo.settings = settings;
321 langInfo.callback = callback;
322
323 LOGFONT font;
324 /* Enumerate all fonts. */
325 font.lfCharSet = DEFAULT_CHARSET;
326 font.lfFaceName[0] = '\0';
327 font.lfPitchAndFamily = 0;
328
329 HDC dc = GetDC(nullptr);
330 int ret = EnumFontFamiliesEx(dc, &font, (FONTENUMPROC)&EnumFontCallback, (LPARAM)&langInfo, 0);
331 ReleaseDC(nullptr, dc);
332 return ret == 0;
333 }

348 static bool TryLoadFontFromFile(const std::string &font_name, LOGFONT &logfont)
349 {
350 wchar_t fontPath[MAX_PATH] = {};
351
352 /* See if this is an absolute path. */
353 if (FileExists(font_name)) {
354 convert_to_fs(font_name, fontPath);
355 } else {
356 /* Scan the search-paths to see if it can be found. */
357 std::string full_font = FioFindFullPath(BASE_DIR, font_name);
358 if (!full_font.empty()) {
359 convert_to_fs(font_name, fontPath);
360 }
361 }
362
363 if (fontPath[0] != 0) {
364 if (AddFontResourceEx(fontPath, FR_PRIVATE, 0) != 0) {
365 /* Try a nice little undocumented function first for getting the internal font name.
366 * Some documentation is found at: http://www.undocprint.org/winspool/getfontresourceinfo */
367 static LibraryLoader _gdi32("gdi32.dll");
368 typedef BOOL(WINAPI *PFNGETFONTRESOURCEINFO)(LPCTSTR, LPDWORD, LPVOID, DWORD);
369 static PFNGETFONTRESOURCEINFO GetFontResourceInfo = _gdi32.GetFunction("GetFontResourceInfoW");
370
371 if (GetFontResourceInfo != nullptr) {
372 /* Try to query an array of LOGFONTs that describe the file. */
373 DWORD len = 0;
374 if (GetFontResourceInfo(fontPath, &len, nullptr, 2) && len >= sizeof(LOGFONT)) {
375 LOGFONT *buf = (LOGFONT *)new uint8_t[len];
376 if (GetFontResourceInfo(fontPath, &len, buf, 2)) {
377 logfont = *buf; // Just use first entry.
378 }
379 delete[](uint8_t *)buf;
380 }
381 }
382
383 /* No dice yet. Use the file name as the font face name, hoping it matches. */
384 if (logfont.lfFaceName[0] == 0) {
385 wchar_t fname[_MAX_FNAME];
386 _wsplitpath(fontPath, nullptr, nullptr, fname, nullptr);
387
388 wcsncpy_s(logfont.lfFaceName, lengthof(logfont.lfFaceName), fname, _TRUNCATE);
389 logfont.lfWeight = StrContainsIgnoreCase(font_name, " bold") || StrContainsIgnoreCase(font_name, "-bold") ? FW_BOLD : FW_NORMAL; // Poor man's way to allow selecting bold fonts.
390 }
391 }
392 }
393
394 return logfont.lfFaceName[0] != 0;
395 }
