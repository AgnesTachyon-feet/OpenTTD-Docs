20/* static */ AIConfig *AIConfig::GetConfig(CompanyID company, ScriptSettingSource source)
21{
22 assert(company < MAX_COMPANIES);
23
24 if (source == SSS_DEFAULT && _game_mode == GM_MENU) source = SSS_FORCE_NEWGAME;
25
26 if (source == SSS_DEFAULT) {
27 Company *c = Company::GetIfValid(company);
28 if (c != nullptr && c->ai_config != nullptr) return c->ai_config.get();
29 }
30
31 auto &config = (source == SSS_FORCE_NEWGAME) ? _settings_newgame.script_config.ai[company] : _settings_game.script_config.ai[company];
32 if (config == nullptr) config = std::make_unique<AIConfig>();
33
34 return config.get();
35}
