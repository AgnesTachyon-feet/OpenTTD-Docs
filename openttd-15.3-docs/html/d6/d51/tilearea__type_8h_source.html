168class OrthogonalTileIterator : public TileIterator {
169private:
170 int w;
171 int x;
172 int y;
173
174public:
179 OrthogonalTileIterator(const OrthogonalTileArea &ta) : TileIterator(ta.w == 0 || ta.h == 0 ? INVALID_TILE : ta.tile), w(ta.w), x(ta.w), y(ta.h)
180 {
181 }
182
188 OrthogonalTileIterator(TileIndex corner1, TileIndex corner2)
189 {
190 *this = OrthogonalTileIterator(OrthogonalTileArea(corner1, corner2));
191 }
192
196 inline TileIterator& operator ++() override
197 {
198 assert(this->tile != INVALID_TILE);
199
200 if (--this->x > 0) {
201 this->tile++;
202 } else if (--this->y > 0) {
203 this->x = this->w;
204 this->tile += TileDiffXY(1 - this->w, 1);
205 } else {
206 this->tile = INVALID_TILE;
207 }
208 return *this;
209 }
210
211 std::unique_ptr<TileIterator> Clone() const override
212 {
213 return std::make_unique<OrthogonalTileIterator>(*this);
214 }
215};
