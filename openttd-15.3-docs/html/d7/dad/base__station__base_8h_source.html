210template <class T, bool Tis_waypoint>
211struct SpecializedStation : public BaseStation {
212 static constexpr StationFacilities EXPECTED_FACIL = Tis_waypoint ? StationFacility::Waypoint : StationFacilities{};
213
218 inline SpecializedStation(TileIndex tile) :
219 BaseStation(tile)
220 {
221 this->facilities = EXPECTED_FACIL;
222 }
223
229 static inline bool IsExpected(const BaseStation *st)
230 {
231 return st->facilities.Test(StationFacility::Waypoint) == Tis_waypoint;
232 }
233
239 static inline bool IsValidID(auto index)
240 {
241 return BaseStation::IsValidID(index) && IsExpected(BaseStation::Get(index));
242 }
243
248 static inline T *Get(auto index)
249 {
250 return (T *)BaseStation::Get(index);
251 }
252
257 static inline T *GetIfValid(auto index)
258 {
259 return IsValidID(index) ? Get(index) : nullptr;
260 }
261
267 static inline T *GetByTile(TileIndex tile)
268 {
269 return GetIfValid(GetStationIndex(tile));
270 }
271
277 static inline T *From(BaseStation *st)
278 {
279 assert(IsExpected(st));
280 return (T *)st;
281 }
282
288 static inline const T *From(const BaseStation *st)
289 {
290 assert(IsExpected(st));
291 return (const T *)st;
292 }
293
299 static Pool::IterateWrapper<T> Iterate(size_t from = 0) { return Pool::IterateWrapper<T>(from); }
