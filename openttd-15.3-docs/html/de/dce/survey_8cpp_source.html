146static void SurveySettingsTable(nlohmann::json &survey, const SettingTable &table, void *object, bool skip_if_default)
147{
148 for (auto &desc : table) {
149 const SettingDesc *sd = GetSettingDesc(desc);
150 /* Skip any old settings we no longer save/load. */
151 if (!SlIsObjectCurrentlyValid(sd->save.version_from, sd->save.version_to)) continue;
152
153 const auto &name = sd->GetName();
154 if (skip_if_default && sd->IsDefaultValue(object)) continue;
155 survey[name] = sd->FormatValue(object);
156 }
157}

312void SurveyCompanies(nlohmann::json &survey)
313{
314 for (const Company *c : Company::Iterate()) {
315 auto &company = survey[fmt::format("{}", c->index.base())];
316 if (c->ai_info == nullptr) {
317 company["type"] = "human";
318 } else {
319 company["type"] = "ai";
320 company["script"] = fmt::format("{}.{}", c->ai_info->GetName(), c->ai_info->GetVersion());
321 }
322
323 for (VehicleType type = VEH_BEGIN; type < VEH_COMPANY_END; type++) {
324 uint amount = c->group_all[type].num_vehicle;
325 company["vehicles"][_vehicle_type_to_string[type]] = amount;
326 }
327
328 company["infrastructure"]["road"] = c->infrastructure.GetRoadTotal();
329 company["infrastructure"]["tram"] = c->infrastructure.GetTramTotal();
330 company["infrastructure"]["rail"] = c->infrastructure.GetRailTotal();
331 company["infrastructure"]["signal"] = c->infrastructure.signal;
332 company["infrastructure"]["water"] = c->infrastructure.water;
333 company["infrastructure"]["station"] = c->infrastructure.station;
334 company["infrastructure"]["airport"] = c->infrastructure.airport;
335 }

359void SurveyGrfs(nlohmann::json &survey)
360{
361 for (const auto &c : _grfconfig) {
362 auto grfid = fmt::format("{:08x}", std::byteswap(c->ident.grfid));
363 auto &grf = survey[std::move(grfid)];
364
365 grf["md5sum"] = FormatArrayAsHex(c->ident.md5sum);
366 grf["status"] = c->status;
367
368 if ((c->palette & GRFP_GRF_MASK) == GRFP_GRF_UNSET) grf["palette"] = "unset";
369 if ((c->palette & GRFP_GRF_MASK) == GRFP_GRF_DOS) grf["palette"] = "dos";
370 if ((c->palette & GRFP_GRF_MASK) == GRFP_GRF_WINDOWS) grf["palette"] = "windows";
371 if ((c->palette & GRFP_GRF_MASK) == GRFP_GRF_ANY) grf["palette"] = "any";
372
373 if ((c->palette & GRFP_BLT_MASK) == GRFP_BLT_UNSET) grf["blitter"] = "unset";
374 if ((c->palette & GRFP_BLT_MASK) == GRFP_BLT_32BPP) grf["blitter"] = "32bpp";
375
376 grf["is_static"] = c->flags.Test(GRFConfigFlag::Static);
377 grf["parameters"] = std::span<const uint32_t>(c->param);
378 }
379}

428 survey["icu_i18n"] = buf;

488std::string SurveyMemoryToText(uint64_t memory)
489{
490 memory = memory / 1024; // KiB
491 memory = CeilDiv(memory, 1024); // MiB
492
493 /* Anything above 512 MiB we represent in GiB. */
494 if (memory > 512) {
495 return fmt::format("{} GiB", CeilDiv(memory, 1024));
496 }
497
498 /* Anything above 64 MiB we represent in a multiplier of 128 MiB. */
499 if (memory > 64) {
500 return fmt::format("{} MiB", Ceil(memory, 128));
501 }
502
503 /* Anything else in a multiplier of 4 MiB. */
504 return fmt::format("{} MiB", Ceil(memory, 4));
505}
