34static EngineID GetNextArticulatedPart(uint index, EngineID front_type, Vehicle *front = nullptr, bool *mirrored = nullptr)
35{
36 assert(front == nullptr || front->engine_type == front_type);
37
38 const Engine *front_engine = Engine::Get(front_type);
39
40 uint16_t callback = GetVehicleCallback(CBID_VEHICLE_ARTIC_ENGINE, index, 0, front_type, front);
41 if (callback == CALLBACK_FAILED) return EngineID::Invalid();
42
43 if (front_engine->GetGRF()->grf_version < 8) {
44 /* 8 bits, bit 7 for mirroring */
45 callback = GB(callback, 0, 8);
46 if (callback == 0xFF) return EngineID::Invalid();
47 if (mirrored != nullptr) *mirrored = HasBit(callback, 7);
48 callback = GB(callback, 0, 7);
49 } else {
50 /* 15 bits, bit 14 for mirroring */
51 if (callback == 0x7FFF) return EngineID::Invalid();
52 if (mirrored != nullptr) *mirrored = HasBit(callback, 14);
53 callback = GB(callback, 0, 14);
54 }
55
56 return GetNewEngineID(front_engine->GetGRF(), front_engine->type, callback);
57}

75uint CountArticulatedParts(EngineID engine_type, bool purchase_window)
76{
77 if (!EngInfo(engine_type)->callback_mask.Test(VehicleCallbackMask::ArticEngine)) return 0;
78
79 /* If we can't allocate a vehicle now, we can't allocate it in the command
80 * either, so it doesn't matter how many articulated parts there are. */
81 if (!Vehicle::CanAllocateItem()) return 0;
82
83 std::unique_ptr<Vehicle> v;
84 if (!purchase_window) {
85 v = std::make_unique<Vehicle>();
86 v->engine_type = engine_type;
87 v->owner = _current_company;
88 }
89
90 uint i;
91 for (i = 1; i < MAX_ARTICULATED_PARTS; i++) {
92 if (GetNextArticulatedPart(i, engine_type, v.get()) == EngineID::Invalid()) break;
93 }
94
95 return i - 1;
96}

136CargoArray GetCapacityOfArticulatedParts(EngineID engine)
137{
138 CargoArray capacity{};
139 const Engine *e = Engine::Get(engine);
140
141 if (auto [cargo, cap] = GetVehicleDefaultCapacity(engine); IsValidCargoType(cargo)) {
142 capacity[cargo] = cap;
143 }
144
145 if (!e->IsGroundVehicle()) return capacity;
146
147 if (!e->info.callback_mask.Test(VehicleCallbackMask::ArticEngine)) return capacity;
148
149 for (uint i = 1; i < MAX_ARTICULATED_PARTS; i++) {
150 EngineID artic_engine = GetNextArticulatedPart(i, engine);
151 if (artic_engine == EngineID::Invalid()) break;
152
153 if (auto [cargo, cap] = GetVehicleDefaultCapacity(artic_engine); IsValidCargoType(cargo)) {
154 capacity[cargo] += cap;
155 }
156 }
157
158 return capacity;
159}

196bool IsArticulatedVehicleRefittable(EngineID engine)
197{
198 if (IsEngineRefittable(engine)) return true;
199
200 const Engine *e = Engine::Get(engine);
201 if (!e->IsGroundVehicle()) return false;
202
203 if (!e->info.callback_mask.Test(VehicleCallbackMask::ArticEngine)) return false;
204
205 for (uint i = 1; i < MAX_ARTICULATED_PARTS; i++) {
206 EngineID artic_engine = GetNextArticulatedPart(i, engine);
207 if (artic_engine == EngineID::Invalid()) break;
208
209 if (IsEngineRefittable(artic_engine)) return true;
210 }
211
212 return false;
213}

222void GetArticulatedRefitMasks(EngineID engine, bool include_initial_cargo_type, CargoTypes *union_mask, CargoTypes *intersection_mask)
223{
224 const Engine *e = Engine::Get(engine);
225 CargoTypes veh_cargoes = GetAvailableVehicleCargoTypes(engine, include_initial_cargo_type);
226 *union_mask = veh_cargoes;
227 *intersection_mask = (veh_cargoes != 0) ? veh_cargoes : ALL_CARGOTYPES;
228
229 if (!e->IsGroundVehicle()) return;
230 if (!e->info.callback_mask.Test(VehicleCallbackMask::ArticEngine)) return;
231
232 for (uint i = 1; i < MAX_ARTICULATED_PARTS; i++) {
233 EngineID artic_engine = GetNextArticulatedPart(i, engine);
234 if (artic_engine == EngineID::Invalid()) break;
235
236 veh_cargoes = GetAvailableVehicleCargoTypes(artic_engine, include_initial_cargo_type);
237 *union_mask |= veh_cargoes;
238 if (veh_cargoes != 0) *intersection_mask &= veh_cargoes;
239 }
240}

262CargoTypes GetCargoTypesOfArticulatedVehicle(const Vehicle *v, CargoType *cargo_type)
263{
264 CargoTypes cargoes = 0;
265 CargoType first_cargo = INVALID_CARGO;
266
267 do {
268 if (IsValidCargoType(v->cargo_type) && v->GetEngine()->CanCarryCargo()) {
269 SetBit(cargoes, v->cargo_type);
270 if (!IsValidCargoType(first_cargo)) first_cargo = v->cargo_type;
271 if (first_cargo != v->cargo_type) {
272 if (cargo_type != nullptr) {
273 *cargo_type = INVALID_CARGO;
274 cargo_type = nullptr;
275 }
276 }
277 }
278
279 v = v->HasArticulatedPart() ? v->GetNextArticulatedPart() : nullptr;
280 } while (v != nullptr);
281
282 if (cargo_type != nullptr) *cargo_type = first_cargo;
283 return cargoes;
284}

294void CheckConsistencyOfArticulatedVehicle(const Vehicle *v)
295{
296 const Engine *engine = v->GetEngine();
297
298 CargoTypes purchase_refit_union, purchase_refit_intersection;
299 GetArticulatedRefitMasks(v->engine_type, true, &purchase_refit_union, &purchase_refit_intersection);
300 CargoArray purchase_default_capacity = GetCapacityOfArticulatedParts(v->engine_type);
301
302 CargoTypes real_refit_union = 0;
303 CargoTypes real_refit_intersection = ALL_CARGOTYPES;
304 CargoTypes real_default_cargoes = 0;
305
306 do {
307 CargoTypes refit_mask = GetAvailableVehicleCargoTypes(v->engine_type, true);
308 real_refit_union |= refit_mask;
309 if (refit_mask != 0) real_refit_intersection &= refit_mask;
310
311 assert(v->cargo_type < NUM_CARGO);
312 if (v->cargo_cap > 0) SetBit(real_default_cargoes, v->cargo_type);
313
314 v = v->HasArticulatedPart() ? v->GetNextArticulatedPart() : nullptr;
315 } while (v != nullptr);
316
317 /* Check whether the vehicle carries more cargoes than expected */
318 bool carries_more = false;
319 for (CargoType cargo_type : SetCargoBitIterator(real_default_cargoes)) {
320 if (purchase_default_capacity[cargo_type] == 0) {
321 carries_more = true;
322 break;
323 }
324 }
325
326 /* show a warning once for each GRF after each game load */
327 if (real_refit_union != purchase_refit_union || real_refit_intersection != purchase_refit_intersection || carries_more) {
328 ShowNewGrfVehicleError(engine->index, STR_NEWGRF_BUGGY, STR_NEWGRF_BUGGY_ARTICULATED_CARGO, GRFBug::VehRefit, false);
329 }
330}

336void AddArticulatedParts(Vehicle *first)
337{
338 VehicleType type = first->type;
339 if (!EngInfo(first->engine_type)->callback_mask.Test(VehicleCallbackMask::ArticEngine)) return;
340
341 Vehicle *v = first;
342 for (uint i = 1; i < MAX_ARTICULATED_PARTS; i++) {
343 bool flip_image;
344 EngineID engine_type = GetNextArticulatedPart(i, first->engine_type, first, &flip_image);
345 if (engine_type == EngineID::Invalid()) return;
346
347 /* In the (very rare) case the GRF reported wrong number of articulated parts
348 * and we run out of available vehicles, bail out. */
349 if (!Vehicle::CanAllocateItem()) return;
350
351 GroundVehicleCache *gcache = v->GetGroundVehicleCache();
352 gcache->first_engine = v->engine_type; // Needs to be set before first callback
353
354 const Engine *e_artic = Engine::Get(engine_type);
355 switch (type) {
356 default: NOT_REACHED();
357
358 case VEH_TRAIN: {
359 Train *front = Train::From(first);
360 Train *t = new Train();
361 v->SetNext(t);
362 v = t;
363
364 t->subtype = 0;
365 t->track = front->track;
366 t->railtypes = front->railtypes;
367
368 t->spritenum = e_artic->VehInfo<RailVehicleInfo>().image_index;
369 if (e_artic->CanCarryCargo()) {
370 t->cargo_type = e_artic->GetDefaultCargoType();
371 t->cargo_cap = e_artic->VehInfo<RailVehicleInfo>().capacity; // Callback 36 is called when the consist is finished
372 } else {
373 t->cargo_type = front->cargo_type; // Needed for livery selection
374 t->cargo_cap = 0;
375 }
376 t->refit_cap = 0;
377
378 t->SetArticulatedPart();
379 break;
380 }
381
382 case VEH_ROAD: {
383 RoadVehicle *front = RoadVehicle::From(first);
384 RoadVehicle *rv = new RoadVehicle();
385 v->SetNext(rv);
386 v = rv;
387
388 rv->subtype = 0;
389 gcache->cached_veh_length = VEHICLE_LENGTH; // Callback is called when the consist is finished
390 rv->state = RVSB_IN_DEPOT;
391
392 rv->roadtype = front->roadtype;
393 rv->compatible_roadtypes = front->compatible_roadtypes;
394
395 rv->spritenum = e_artic->VehInfo<RoadVehicleInfo>().image_index;
396 if (e_artic->CanCarryCargo()) {
397 rv->cargo_type = e_artic->GetDefaultCargoType();
398 assert(IsValidCargoType(rv->cargo_type));
399 rv->cargo_cap = e_artic->VehInfo<RoadVehicleInfo>().capacity; // Callback 36 is called when the consist is finished
400 } else {
401 rv->cargo_type = front->cargo_type; // Needed for livery selection
402 rv->cargo_cap = 0;
403 }
404 rv->refit_cap = 0;
405
406 rv->SetArticulatedPart();
407 break;
408 }
409 }
410
411 /* get common values from first engine */
412 v->direction = first->direction;
413 v->owner = first->owner;
414 v->tile = first->tile;
415 v->x_pos = first->x_pos;
416 v->y_pos = first->y_pos;
417 v->z_pos = first->z_pos;
418 v->date_of_last_service = first->date_of_last_service;
419 v->date_of_last_service_newgrf = first->date_of_last_service_newgrf;
420 v->build_year = first->build_year;
421 v->vehstatus = first->vehstatus;
422 v->vehstatus.Reset(VehState::Stopped);
423
424 v->cargo_subtype = 0;
425 v->max_age = CalendarTime::MIN_DATE;
426 v->engine_type = engine_type;
427 v->value = 0;
428 v->sprite_cache.sprite_seq.Set(SPR_IMG_QUERY);
429 v->random_bits = Random();
430
431 if (flip_image) v->spritenum++;
432
433 if (v->type == VEH_TRAIN) {
434 auto prob = TestVehicleBuildProbability(v, v->engine_type, BuildProbabilityType::Reversed);
435 if (prob.has_value()) Train::From(v)->flags.Set(VehicleRailFlag::Flipped, prob.value());
436 }
437 v->UpdatePosition();
438 }
