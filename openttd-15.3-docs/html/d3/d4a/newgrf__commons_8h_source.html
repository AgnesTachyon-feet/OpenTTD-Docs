75inline uint GetConstructionStageOffset(uint construction_stage, uint num_sprites)
76{
77 assert(num_sprites > 0);
78 if (num_sprites > 4) num_sprites = 4;
79 switch (construction_stage) {
80 case 0: return 0;
81 case 1: return num_sprites > 2 ? 1 : 0;
82 case 2: return num_sprites > 2 ? num_sprites - 2 : 0;
83 case 3: return num_sprites - 1;
84 default: NOT_REACHED();
85 }
86}

405template <class Tkey>
406struct VariableGRFFileProps : GRFFilePropsBase {
407 using ValueType = std::pair<Tkey, const struct SpriteGroup *>;
408 std::vector<ValueType> spritegroups;
409
415 const SpriteGroup *GetSpriteGroup(Tkey index) const
416 {
417 auto it = std::ranges::lower_bound(this->spritegroups, index, std::less{}, &ValueType::first);
418 if (it == std::end(this->spritegroups) || it->first != index) return nullptr;
419 return it->second;
420 }
421
427 const struct SpriteGroup *GetFirstSpriteGroupOf(std::initializer_list<Tkey> indices) const
428 {
429 for (auto key : indices) {
430 auto *result = GetSpriteGroup(key);
431 if (result != nullptr) return result;
432 }
433 return nullptr;
434 }
435
441 void SetSpriteGroup(Tkey index, const SpriteGroup *spritegroup)
442 {
443 auto it = std::ranges::lower_bound(this->spritegroups, index, std::less{}, &ValueType::first);
444 if (it == std::end(this->spritegroups) || it->first != index) {
445 this->spritegroups.emplace(it, index, spritegroup);
446 } else {
447 it->second = spritegroup;
448 }
449 }
450};
